    61→// ============ MAIN COMPONENT ============
    62→export default function StellarisSaveEditor() {
    63→  const { toast } = useToast();
    64→  const fileInputRef = useRef<HTMLInputElement>(null);
    65→
    66→  // State
    67→  const [loading, setLoading] = useState(false);
    68→  const [loaded, setLoaded] = useState(false);
    69→  const [meta, setMeta] = useState<SaveMeta | null>(null);
    70→  const [stats, setStats] = useState<GameStats | null>(null);
    71→  const [resources, setResources] = useState<Record<string, ResourceInfo>>({});
    72→  const [resourceCategories, setResourceCategories] = useState<ResourceCategory>({});
    73→  const [countries, setCountries] = useState<CountryInfo[]>([]);
    74→  const [species, setSpecies] = useState<SpeciesInfo[]>([]);
    75→  const [selectedCountry, setSelectedCountry] = useState('0');
    76→  const [editName, setEditName] = useState('');
    77→  const [editDate, setEditDate] = useState('');
    78→  const [editResources, setEditResources] = useState<Record<string, string>>({});
    79→  const [saving, setSaving] = useState(false);
    80→  const [exporting, setExporting] = useState(false);
    81→  const [activeTab, setActiveTab] = useState('overview');
    82→
    83→  // Load save file
    84→  const handleFileUpload = useCallback(async (file: File) => {
    85→    if (!file.name.endsWith('.sav')) {
    86→      toast({ title: '错误', description: '请选择 .sav 格式的群星存档文件', variant: 'destructive' });
    87→      return;
    88→    }
    89→    setLoading(true);
    90→    try {
    91→      const result = await uploadSave(file);
    92→      setMeta(result.meta);
    93→      setEditName(result.meta.name);
    94→      setEditDate(result.meta.date);
    95→      setSelectedCountry(result.player_country_id);
    96→      setLoaded(true);
    97→      setActiveTab('overview');
    98→      toast({ title: '存档加载成功', description: `${result.meta.name} (${result.meta.date})` });
    99→
   100→      // Load additional data
   101→      const [statsData, countriesData, resourcesData, speciesData] = await Promise.all([
   102→        getStats(),
   103→        getCountries(),
   104→        getResources(result.player_country_id),
   105→        getSpecies(),
   106→      ]);
   107→      setStats(statsData);
   108→      setCountries(countriesData.countries);
   109→      setResources(resourcesData.resources);
   110→      setResourceCategories(resourcesData.categories);
   111→      setSpecies(speciesData.species);
   112→
   113→      // Init edit resources
   114→      const initEdit: Record<string, string> = {};
   115→      for (const [k, v] of Object.entries(resourcesData.resources)) {
   116→        initEdit[k] = v.value.toFixed(2);
   117→      }
   118→      setEditResources(initEdit);
   119→    } catch (err) {
   120→      toast({
   121→        title: '加载失败',
   122→        description: err instanceof Error ? err.message : '未知错误',
   123→        variant: 'destructive',
   124→      });
   125→    } finally {
   126→      setLoading(false);
   127→    }
   128→  }, [toast]);
   129→
   130→  // Handle drag & drop
   131→  const [dragOver, setDragOver] = useState(false);
   132→
   133→  const handleDrop = useCallback((e: React.DragEvent) => {
   134→    e.preventDefault();
   135→    setDragOver(false);
   136→    const file = e.dataTransfer.files[0];
   137→    if (file) handleFileUpload(file);
   138→  }, [handleFileUpload]);
   139→
   140→  // Save resources
   141→  const handleSaveResources = async () => {
   142→    setSaving(true);
   143→    try {
   144→      const parsed: Record<string, number> = {};
   145→      for (const [k, v] of Object.entries(editResources)) {
   146→        const num = parseFloat(v);
   147→        if (!isNaN(num)) parsed[k] = num;
   148→      }
   149→      await updateResources(selectedCountry, parsed);
   150→      // Refresh resources
   151→      const data = await getResources(selectedCountry);
   152→      setResources(data.resources);
   153→      const initEdit: Record<string, string> = {};
   154→      for (const [k, v] of Object.entries(data.resources)) {
   155→        initEdit[k] = v.value.toFixed(2);
   156→      }
   157→      setEditResources(initEdit);
   158→      toast({ title: '保存成功', description: '资源已更新' });
   159→    } catch (err) {
   160→      toast({ title: '保存失败', description: err instanceof Error ? err.message : '未知错误', variant: 'destructive' });
   161→    } finally {
   162→      setSaving(false);
   163→    }
   164→  };
   165→
   166→  // Save date
   167→  const handleSaveDate = async () => {
   168→    if (!/^\d{4}\.\d{2}\.\d{2}$/.test(editDate)) {
   169→      toast({ title: '格式错误', description: '日期格式应为 YYYY.MM.DD', variant: 'destructive' });
   170→      return;
   171→    }
   172→    setSaving(true);
   173→    try {
   174→      await updateDate(editDate);
   175→      const newStats = await getStats();
   176→      setStats(newStats);
   177→      if (meta) setMeta({ ...meta, date: editDate });
   178→      toast({ title: '日期已更新', description: editDate });
   179→    } catch (err) {
   180→      toast({ title: '更新失败', description: err instanceof Error ? err.message : '未知错误', variant: 'destructive' });
   181→    } finally {
   182→      setSaving(false);
   183→    }
   184→  };
   185→
   186→  // Save name
   187→  const handleSaveName = async () => {
   188→    if (!editName.trim()) {
   189→      toast({ title: '名称不能为空', variant: 'destructive' });
   190→      return;
   191→    }
   192→    setSaving(true);
   193→    try {
   194→      await updateName(editName.trim());
   195→      if (meta) setMeta({ ...meta, name: editName.trim() });
   196→      const newCountries = await getCountries();
   197→      setCountries(newCountries.countries);
   198→      toast({ title: '名称已更新', description: editName.trim() });
   199→    } catch (err) {
   200→      toast({ title: '更新失败', description: err instanceof Error ? err.message : '未知错误', variant: 'destructive' });
   201→    } finally {
   202→      setSaving(false);
   203→    }
   204→  };
   205→
   206→  // Export
   207→  const handleExport = async () => {
   208→    setExporting(true);
   209→    try {
   210→      const blob = await exportSave();
   211→      const url = URL.createObjectURL(blob);
   212→      const a = document.createElement('a');
   213→      a.href = url;
   214→      a.download = meta ? `${meta.name.replace(/[^a-zA-Z0-9\u4e00-\u9fa5]/g, '_')}_modified.sav` : 'modified_save.sav';
   215→      document.body.appendChild(a);
   216→      a.click();
   217→      document.body.removeChild(a);
   218→      URL.revokeObjectURL(url);
   219→      toast({ title: '导出成功', description: '修改后的存档已下载' });
   220→    } catch (err) {
   221→      toast({ title: '导出失败', description: err instanceof Error ? err.message : '未知错误', variant: 'destructive' });
   222→    } finally {
   223→      setExporting(false);
   224→    }
   225→  };
   226→
   227→  // Reset
   228→  const handleReset = async () => {
   229→    setLoaded(false);
   230→    setMeta(null);
   231→    setStats(null);
   232→    setResources({});
   233→    setCountries([]);
   234→    setSpecies([]);
   235→    setEditResources({});
   236→    try { await releaseSave(); } catch { /* ignore */ }
   237→    toast({ title: '存档已释放' });
   238→  };
   239→
   240→  // Switch country
   241→  const handleCountryChange = async (countryId: string) => {
   242→    setSelectedCountry(countryId);
   243→    try {
   244→      const data = await getResources(countryId);
   245→      setResources(data.resources);
   246→      setResourceCategories(data.categories);
   247→      const initEdit: Record<string, string> = {};
   248→      for (const [k, v] of Object.entries(data.resources)) {
   249→        initEdit[k] = v.value.toFixed(2);
   250→      }
   251→      setEditResources(initEdit);
   252→    } catch (err) {
   253→      toast({ title: '加载资源失败', description: err instanceof Error ? err.message : '', variant: 'destructive' });
   254→    }
   255→  };
   256→
   257→  // Quick set resource value
   258→  const setResourceValue = (key: string, value: string) => {
   259→    setEditResources(prev => ({ ...prev, [key]: value }));
   260→  };
   261→
   262→  // ============ RENDER: UPLOAD SCREEN ============
   263→  if (!loaded) {
   264→    return (
   265→      <div className="min-h-screen flex items-center justify-center p-4"
   266→        style={{ background: 'radial-gradient(ellipse at 20% 50%, rgba(30,10,60,0.8) 0%, rgba(5,5,15,1) 70%)' }}
   267→      >
   268→        <div className="absolute inset-0 overflow-hidden pointer-events-none">
   269→          {/* Stars background */}
   270→          {Array.from({ length: 80 }).map((_, i) => (
   271→            <div
   272→              key={i}
   273→              className="absolute rounded-full bg-white"
   274→              style={{
   275→                width: Math.random() * 2 + 1,
   276→                height: Math.random() * 2 + 1,
   277→                top: `${Math.random() * 100}%`,
   278→                left: `${Math.random() * 100}%`,
   279→                opacity: Math.random() * 0.7 + 0.3,
   280→                animation: `twinkle ${2 + Math.random() * 3}s ease-in-out ${Math.random() * 2}s infinite`,
   281→              }}
   282→            />
   283→          ))}
   284→        </div>
   285→
   286→        <Card className="relative z-10 w-full max-w-lg border-amber-500/20 bg-black/60 backdrop-blur-xl"
   287→          style={{ boxShadow: '0 0 80px rgba(245,158,11,0.1)' }}
   288→        >
   289→          <CardHeader className="text-center pb-2">
   290→            <div className="mx-auto mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-amber-500/10 border border-amber-500/30">
   291→              <Star className="h-8 w-8 text-amber-400" />
   292→            </div>
   293→            <CardTitle className="text-2xl font-bold text-amber-100">
   294→              群星存档修改器
   295→            </CardTitle>
   296→            <CardDescription className="text-amber-200/60 mt-1">
   297→              Stellaris Save Editor  ·  支持 Caelum / 最新版本存档
   298→            </CardDescription>
   299→          </CardHeader>
   300→          <CardContent>
   301→            <div
   302→              className={`border-2 border-dashed rounded-xl p-10 text-center transition-all cursor-pointer ${
   303→                dragOver
   304→                  ? 'border-amber-400 bg-amber-500/10'
   305→                  : 'border-gray-600 hover:border-amber-500/50 hover:bg-white/5'
   306→              }`}
   307→              onDragOver={(e) => { e.preventDefault(); setDragOver(true); }}
   308→              onDragLeave={() => setDragOver(false)}
   309→              onDrop={handleDrop}
   310→              onClick={() => fileInputRef.current?.click()}
   311→            >
   312→              <input
   313→                ref={fileInputRef}
   314→                type="file" accept=".sav" className="hidden"
   315→                onChange={(e) => {
   316→                  const file = e.target.files?.[0];
   317→                  if (file) handleFileUpload(file);
   318→                }}
   319→              />
   320→              {loading ? (
   321→                <div className="flex flex-col items-center gap-3">
   322→                  <Loader2 className="h-10 w-10 text-amber-400 animate-spin" />
   323→                  <p className="text-amber-200">正在解析存档，请稍候...</p>
   324→                  <p className="text-xs text-amber-200/40">大型存档可能需要10-20秒</p>
   325→                </div>
   326→              ) : (
   327→                <div className="flex flex-col items-center gap-3">
   328→                  <Upload className="h-10 w-10 text-amber-400/60" />
   329→                  <p className="text-amber-100 font-medium">拖拽 .sav 存档文件到此处</p>
   330→                  <p className="text-sm text-amber-200/50">或点击选择文件</p>
   331→                </div>
   332→              )}
   333→            </div>
   334→            <div className="mt-4 flex items-center gap-2 text-xs text-amber-200/30 justify-center">
   335→              <Shield className="h-3 w-3" />
   336→              <span>所有解析均在本地完成，不会上传任何数据</span>
   337→            </div>
   338→          </CardContent>
   339→        </Card>
   340→      </div>
   341→    );
   342→  }
   343→
   344→  // ============ RENDER: EDITOR SCREEN ============
   345→  return (
   346→    <div className="min-h-screen flex flex-col"
   347→      style={{ background: 'linear-gradient(135deg, rgba(5,5,20,1) 0%, rgba(15,10,30,1) 50%, rgba(5,10,25,1) 100%)' }}
   348→    >
   349→      {/* Header */}
   350→      <header className="sticky top-0 z-50 border-b border-amber-500/10 bg-black/70 backdrop-blur-xl">
   351→        <div className="max-w-7xl mx-auto px-4 h-14 flex items-center justify-between">
   352→          <div className="flex items-center gap-3">
   353→            <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-amber-500/15">
   354→              <Star className="h-4 w-4 text-amber-400" />
   355→            </div>
   356→            <div>
   357→              <h1 className="text-sm font-bold text-amber-100 leading-tight">{meta?.name || '群星存档修改器'}</h1>
   358→              <p className="text-xs text-amber-200/40">{meta?.version}  ·  {meta?.date}</p>
   359→            </div>
   360→            {meta?.ironman && (
   361→              <Badge variant="outline" className="border-red-500/40 text-red-400 text-xs">
   362→                铁人模式
   363→              </Badge>
   364→            )}
   365→          </div>
   366→          <div className="flex items-center gap-2">
   367→            <Button variant="outline" size="sm" onClick={handleReset}
   368→              className="border-gray-600 text-gray-300 hover:text-white hover:bg-white/5">
   369→              <RotateCcw className="h-3.5 w-3.5 mr-1" /> 释放存档
   370→            </Button>
   371→            <Button size="sm" onClick={handleExport} disabled={exporting}
   372→              className="bg-amber-500 hover:bg-amber-400 text-black font-medium">
   373→              {exporting ? <Loader2 className="h-3.5 w-3.5 mr-1 animate-spin" /> : <Download className="h-3.5 w-3.5 mr-1" />}
   374→              导出存档
   375→            </Button>
   376→          </div>
   377→        </div>
   378→      </header>
   379→
   380→      {/* Main Content */}
   381→      <main className="flex-1 max-w-7xl mx-auto w-full px-4 py-6">
   382→        <Tabs value={activeTab} onValueChange={setActiveTab}>
   383→          <TabsList className="bg-black/40 border border-amber-500/10 mb-6">
   384→            <TabsTrigger value="overview" className="data-[state=active]:bg-amber-500/20 data-[state=active]:text-amber-100">
   385→              <Globe className="h-3.5 w-3.5 mr-1.5" /> 总览
   386→            </TabsTrigger>
   387→            <TabsTrigger value="resources" className="data-[state=active]:bg-amber-500/20 data-[state=active]:text-amber-100">
   388→              <Zap className="h-3.5 w-3.5 mr-1.5" /> 资源
   389→            </TabsTrigger>
   390→            <TabsTrigger value="countries" className="data-[state=active]:bg-amber-500/20 data-[state=active]:text-amber-100">
   391→              <Flag className="h-3.5 w-3.5 mr-1.5" /> 国家
   392→            </TabsTrigger>
   393→            <TabsTrigger value="species" className="data-[state=active]:bg-amber-500/20 data-[state=active]:text-amber-100">
   394→              <Dna className="h-3.5 w-3.5 mr-1.5" /> 物种
   395→            </TabsTrigger>
   396→          </TabsList>
   397→
   398→          {/* ===== OVERVIEW TAB ===== */}
   399→          <TabsContent value="overview" className="space-y-6">
   400→            {/* Empire Info Card */}
   401→            <Card className="border-amber-500/10 bg-black/40 backdrop-blur">
   402→              <CardHeader className="pb-3">
   403→                <CardTitle className="text-amber-100 text-base flex items-center gap-2">
   404→                  <Star className="h-4 w-4" /> 帝国信息
   405→                </CardTitle>
   406→              </CardHeader>
   407→              <CardContent>
   408→                <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   409→                  {/* Name edit */}
   410→                  <div className="space-y-2">
   411→                    <Label className="text-amber-200/70 text-xs">帝国名称</Label>
   412→                    <div className="flex gap-2">
   413→                      <Input value={editName} onChange={e => setEditName(e.target.value)}
   414→                        className="bg-white/5 border-gray-600 text-amber-50" />
   415→                      <Button size="sm" onClick={handleSaveName} disabled={saving}
   416→                        className="bg-amber-500/20 hover:bg-amber-500/30 text-amber-200 border border-amber-500/30 shrink-0">
   417→                        <Save className="h-3.5 w-3.5" />
   418→                      </Button>
   419→                    </div>
   420→                  </div>
   421→                  {/* Date edit */}
   422→                  <div className="space-y-2">
   423→                    <Label className="text-amber-200/70 text-xs">游戏日期</Label>
   424→                    <div className="flex gap-2">
   425→                      <Input value={editDate} onChange={e => setEditDate(e.target.value)}
   426→                        placeholder="YYYY.MM.DD" className="bg-white/5 border-gray-600 text-amber-50" />
   427→                      <Button size="sm" onClick={handleSaveDate} disabled={saving}
   428→                        className="bg-amber-500/20 hover:bg-amber-500/30 text-amber-200 border border-amber-500/30 shrink-0">
   429→                        <Save className="h-3.5 w-3.5" />
   430→                      </Button>
   431→                    </div>
   432→                  </div>
   433→                </div>
   434→              </CardContent>
   435→            </Card>
   436→
   437→            {/* Stats Grid */}
   438→            {stats && (
   439→              <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   440→                {[
   441→                  { label: '游戏日期', value: stats.date, icon: <Calendar className="h-4 w-4 text-blue-400" /> },
   442→                  { label: '帝国规模', value: stats.empire_size, icon: <MapPin className="h-4 w-4 text-green-400" /> },
   443→                  { label: '科技数量', value: stats.tech_count, icon: <BookOpen className="h-4 w-4 text-purple-400" /> },
   444→                  { label: '舰队规模', value: stats.fleet_size, icon: <Rocket className="h-4 w-4 text-red-400" /> },
   445→                  { label: '军事力量', value: Math.round(stats.military_power).toLocaleString(), icon: <Swords className="h-4 w-4 text-orange-400" /> },
   446→                  { label: '国家数量', value: stats.num_countries, icon: <Flag className="h-4 w-4 text-cyan-400" /> },
   447→                  { label: '物种数量', value: stats.num_species, icon: <Dna className="h-4 w-4 text-green-300" /> },
   448→                  { label: '行星数量', value: stats.owned_planets_count, icon: <Globe className="h-4 w-4 text-amber-400" /> },
   449→                ].map(stat => (
   450→                  <Card key={stat.label} className="border-gray-700/50 bg-black/30">
   451→                    <CardContent className="p-4 flex items-center gap-3">
