    75→def extract_save(filepath: str):
    76→    """Extract .sav (ZIP) and read meta + gamestate texts."""
    77→    with zipfile.ZipFile(filepath, 'r') as z:
    78→        names = z.namelist()
    79→        meta_text = z.read('meta').decode('utf-8') if 'meta' in names else ''
    80→        gamestate_text = z.read('gamestate').decode('utf-8') if 'gamestate' in names else ''
    81→    return meta_text, gamestate_text

    84→def find_player_country_id(gamestate: dict) -> int:
    85→    """Find the player's country index."""
    86→    player = gamestate.get('player')
    87→    if not player:
    88→        return 0
    89→    if isinstance(player, dict):
    90→        items = player.get(None, [])
    91→        if items and isinstance(items, list) and len(items) > 0:
    92→            first = items[0]
    93→            if isinstance(first, dict) and 'country' in first:
    94→                return int(first['country'])
    95→    return 0

    99→    """Extract resources for a specific country from modules.standard_economy_module."""
   100→    countries = gamestate.get('country', {})
   101→    c_key = str(country_idx)
   102→    if c_key not in countries:
   103→        return {}
   104→    country = countries[c_key]
   105→    if not isinstance(country, dict):
   106→        return {}
   107→    modules = country.get('modules', {})
   108→    if not isinstance(modules, dict):
   109→        return {}
   110→    econ = modules.get('standard_economy_module', {})
   111→    if not isinstance(econ, dict):
   112→        return {}
   113→    resources = econ.get('resources', {})
   114→    return resources if isinstance(resources, dict) else {}

   176→    """Get budget/income info for a country."""
   177→    countries = gamestate.get('country', {})
   178→    c_key = str(country_idx)
   179→    if c_key not in countries:
   180→        return {}
   181→    country = countries[c_key]
   182→    if not isinstance(country, dict):
   183→        return {}
   184→    budget = country.get('budget', {})
   185→    if not isinstance(budget, dict):
   186→        return {}
   187→    current = budget.get('current_month', {})
   188→    if not isinstance(current, dict):
   189→        return {}
   190→    income = current.get('income', {})
   191→    if not isinstance(income, dict):
   192→        return {}
   193→    result = {}
   194→    for source_name, source_data in income.items():
   195→        if not isinstance(source_data, dict):
   196→            continue
   197→        for res_key, val in source_data.items():
   198→            if res_key not in result:
   199→                result[res_key] = 0.0
   200→            try:
   201→                result[res_key] += float(val)
   202→            except (ValueError, TypeError):
   203→                pass

   228→def _find_block_by_path(text_lines: list, path_steps: list) -> tuple:
   229→    """
   230→    Navigate through nested blocks to find the target block.
   231→    Returns (start_line_idx, end_line_idx, indent_level) or (-1,-1,-1) if not found.
   232→    """
   233→    depth = 0
   234→    step_idx = 0
   235→    current_step = path_steps[0]
   236→    block_start = -1
   237→
   238→    for i, line in enumerate(text_lines):
   239→        stripped = line.strip()
   240→
   241→        # Track depth via braces
   242→        for ch in stripped:
   243→            if ch == '{':
   244→                depth += 1
   245→            elif ch == '}':
   246→                depth -= 1
   247→
   248→        if step_idx < len(path_steps):
   249→ # Check if this line matches the current step
   250→            pattern = re.compile(rf'^\s*{re.escape(str(current_step))}\s*[=\{{]')
   251→            if pattern.match(line):
   252→                if step_idx == 0:
   253→                    block_start = i
   254→                step_idx += 1
   255→                if step_idx < len(path_steps):
   256→                    current_step = path_steps[step_idx]
   257→                else:
   258→                    # Found the target block
   259→                    # Return the block content range (after the opening brace)
   260→                    return (block_start, i, depth)
   261→
   262→    return (-1, -1, -1)

   265→def modify_resource_in_text(text: str, country_idx: int, resource_key: str, new_value: float) -> str:
   266→    """Modify a single resource value directly in the raw gamestate text."""
   267→    lines = text.split('\n')
   268→
   269→    # State machine to navigate: country={ 0={ ... modules={ standard_economy_module={ resources={ resource_key=VALUE } } } } }
   270→    state = 'seeking_country'
   271→    country_depth = 0
   272→    country_idx_found = False
   273→    module_depth = 0
   274→    econ_found = False
   275→    resource_block_depth = 0
   276→    in_resource_block = False
   277→    result_lines = []
   278→    modified = False
   279→
   280→    for line in lines:
   281→        if modified:
   282→            result_lines.append(line)
   283→            continue
   284→
   285→        stripped = line.strip()
   286→
   287→        # Count braces on this line
   288→        opens = stripped.count('{')
   289→        closes = stripped.count('}')
   290→
   291→        if state == 'seeking_country':
   292→            if stripped == 'country={':
   293→                state = 'in_country'
   294→                country_depth = 1
   295→            result_lines.append(line)
   296→            continue
   297→
   298→        if state == 'in_country':
   299→            country_depth += opens - closes
   300→
   301→            if country_depth <= 0:
   302→                state = 'seeking_country'
   303→                result_lines.append(line)
   304→                continue
   305→
   306→            if not country_idx_found:
   307→                # Look for the specific country index
   308→                idx_pat = re.compile(r'^\s*' + re.escape(str(country_idx)) + r'=\{\s*$')
   309→                if idx_pat.match(line):
   310→                    country_idx_found = True
   311→                    result_lines.append(line)
   312→                    continue
   313→
   314→            if country_idx_found and not econ_found:
   315→                # Look for standard_economy_module
   316→                if re.match(r'^\s*standard_economy_module\s*={\s*$', line):
   317→                    econ_found = True
   318→                    result_lines.append(line)
   319→                    continue
   320→
   321→            if econ_found and not in_resource_block:
   322→                if re.match(r'^\s*resources\s*={\s*$', line):
   323→                    in_resource_block = True
   324→                    resource_block_depth = 1
   325→                    result_lines.append(line)
   326→                    continue
   327→
   328→            if in_resource_block:
   329→                resource_block_depth += opens - closes
   330→
   331→                if resource_block_depth <= 0:
   332→                    in_resource_block = False
   333→                    result_lines.append(line)
   334→                    continue
   335→
   336→                # Check for the resource key
   337→                match = re.match(rf'^(\s*){re.escape(resource_key)}=(.+)$', line)
   338→                if match:
   339→                    indent = match.group(1)
   340→                    formatted = f'{new_value:.5f}'.rstrip('0')
   341→                    if formatted.endswith('.'):
   342→                        formatted += '0'
   343→                    result_lines.append(f'{indent}{resource_key}={formatted}')
   344→                    modified = True
   345→                    continue
   346→
   347→            result_lines.append(line)
   348→            continue
   349→
   350→        result_lines.append(line)
   351→
   352→    return '\n'.join(result_lines)

   355→def modify_date_in_text(text: str, new_date: str) -> str:
   356→    """Modify the game date in text."""
   357→    pattern = re.compile(r'^(date=")\d{4}\.\d{2}\.\d{2}(")', re.MULTILINE)
   358→    return pattern.sub(f'\\g<1>{new_date}\\g<2>', text, count=1)

   361→def modify_name_in_text(text: str, old_name: str, new_name: str) -> str:
   362→    """Modify the empire name in gamestate text."""
   363→    # In gamestate, the name is in the first country's name block
   364→    # Format: name={\n\tkey="OLDNAME"\n}
   365→    pattern = re.compile(r'^(name=\{\n\s*key=")' + re.escape(old_name) + r'(")', re.MULTILINE)
   366→    result = pattern.sub(f'\\g<1>{new_name}\\g<2>', text, count=1)
   367→    return result

   370→def modify_name_in_meta(text: str, new_name: str) -> str:
   371→    """Modify the empire name in meta text."""
   372→    pattern = re.compile(r'^(name=")' + r'[^"]*' + r'(")', re.MULTILINE)
   373→    return pattern.sub(f'\\g<1>{new_name}\\g<2>', text, count=1)

   376→class SaveHandler(BaseHTTPRequestHandler):
   377→    """HTTP request handler for save file operations."""

   379→    def _send_json(self, data: dict, status: int = 200):
   380→        body = json.dumps(data, ensure_ascii=False).encode('utf-8')
   381→        self.send_response(status)
   382→        self.send_header('Content-Type', 'application/json; charset=utf-8')
   383→        self.send_header('Access-Control-Allow-Origin', '*')
   384→        self.send_header('Access-Control-Allow-Methods', 'GET, POST, PUT, DELETE, OPTIONS')
   385→        self.send_header('Access-Control-Allow-Headers', 'Content-Type')
   386→        self.send_header('Content-Length', str(len(body)))
   387→        self.end_headers()
   388→        self.wfile.write(body)
   389→
   390→    def _send_file(self, filepath: str, filename: str):
   391→        import mimetypes
   392→        ctype = mimetypes.guess_type(filename)[0] or 'application/octet-stream'
   393→        size = os.path.getsize(filepath)
   394→        self.send_response(200)
   395→        self.send_header('Content-Type', ctype)
   396→        self.send_header('Content-Disposition', f'attachment; filename="{filename}"')
   397→        self.send_header('Access-Control-Allow-Origin', '*')
   398→        self.send_header('Content-Length', str(size))
   399→        self.end_headers()
   400→        with open(filepath, 'rb') as f:
   401→            while True:
   402→                chunk = f.read(65536)
   403→                if not chunk:
   404→                    break
   405→                self.wfile.write(chunk)
   406→
   407→    def _read_body(self) -> bytes:
   408→        length = int(self.headers.get('Content-Length', 0))
   409→        return self.rfile.read(length) if length > 0 else b''

   500→            elif path == '/api/stats':
   501→                if not save_state['gamestate_parsed']:
   502→                    self._send_json({'error': 'No save file loaded'}, 400)
   503→                    return
   504→                gs = save_state['gamestate_parsed']

   507→                c_key = str(player_idx)

   512→                    'num_species': len(gs.get('species_db', {})) if isinstance(gs.get('species_db'), dict) else 0,

   541→                tmp = tempfile.NamedTemporaryFile(suffix='.sav', delete=False)
   542→                tmp.write(body)
   543→                tmp.close()
   544→
   545→                try:
   546→                    meta_text, gamestate_text = extract_save(tmp.name)
   547→                except Exception as e:
   548→                    os.unlink(tmp.name)
   549→                    self._send_json({'error': f'Failed to parse .sav file: {str(e)}'}, 400)
   550→                    return
   551→
   552→                meta_parsed = parse_clausewitz(meta_text) if meta_text else {}
   553→
   554→                gamestate_parsed = {}
   555→                try:
   556→                    gamestate_parsed = parse_clausewitz(gamestate_text) if gamestate_text else {}
   557→                except Exception as e:
   558→                    print(f'Warning: Full gamestate parse failed: {e}')
   559→
   560→                save_state['sav_path'] = tmp.name
   561→                save_state['meta_text'] = meta_text
   562→                save_state['gamestate_text'] = gamestate_text
   563→                save_state['meta_parsed'] = meta_parsed
   564→                save_state['gamestate_parsed'] = gamestate_parsed
   565→
   566→                player_idx = find_player_country_id(gamestate_parsed)
   567→                meta_name = meta_parsed.get('name', '')
   568→                meta_date = meta_parsed.get('date', '')
   569→
   570→                dlcs = meta_parsed.get('required_dlcs', {})
   571→                if isinstance(dlcs, dict):
   572→                    dlcs = dlcs.get(None, [])
   573→                if not isinstance(dlcs, list):
   574→                    dlcs = [dlcs]
   575→
   576→                self._send_json({
   577→                    'success': True,
   578→                    'filename': os.path.basename(tmp.name),
   579→                    'meta': {
   580→                        'version': meta_parsed.get('version', ''),
   581→                        'name': meta_name,
   582→                        'date': meta_date,
   583→                        'ironman': meta_parsed.get('ironman', False),
   584→                        'dlcs': dlcs,
   585→                        'meta_fleets': meta_parsed.get('meta_fleets', 0),
   586→                        'meta_planets': meta_parsed.get('meta_planets', 0),
   587→                    },
   588→                    'player_country_id': str(player_idx),
   589→                    'gamestate_size': len(gamestate_text),
   590→                })
   591→
   592→            elif path == '/api/export':
   593→                if not save_state['sav_path']:
   594→                    self._send_json({'error': 'No save file loaded'}, 400)
   595→                    return
   596→
   597→                # Create modified .sav file
   598→                export_tmp = tempfile.NamedTemporaryFile(suffix='.sav', delete=False)
   599→                export_tmp.close()
   600→
   601→                with zipfile.ZipFile(export_tmp.name, 'w', zipfile.ZIP_DEFLATED) as zout:
   602→                    zout.writestr('meta', save_state['meta_text'].encode('utf-8'))
   603→                    zout.writestr('gamestate', save_state['gamestate_text'].encode('utf-8'))
   604→
   605→                self._send_file(export_tmp.name, 'modified_save.sav')
   606→
   607→                # Clean up after sending
   608→                os.unlink(export_tmp.name)
   609→
   610→            else:
   611→                self._send_json({'error': 'Not found'}, 404)
   612→
   613→        except Exception as e:
   614→            traceback.print_exc()
   615→            self._send_json({'error': str(e)}, 500)

   617→    def do_PUT(self):
   618→        parsed = urlparse(self.path)
   619→        path = parsed.path
   620→
   621→        try:
   622→            body = self._read_body()
   623→            data = json.loads(body) if body else {}
   624→
   625→            if path == '/api/resources':
   626→                if not save_state['gamestate_text']:
   627→                    self._send_json({'error': 'No save file loaded'}, 400)
   628→                    return
   629→
   630→                country_id = data.get('country_id', 0)
   631→                resources = data.get('resources', {})
   632→
   633→                gs_text = save_state['gamestate_text']
   634→                for key, value in resources.items():
   635→                    try:
   636→                        gs_text = modify_resource_in_text(gs_text, int(country_id), key, float(value))
   637→                    except (ValueError, TypeError):
   638→                        pass
   639→
   640→                save_state['gamestate_text'] = gs_text
   641→
   642→                # Re-parse to update cache
   643→                try:
   644→                    save_state['gamestate_parsed'] = parse_clausewitz(gs_text)
   645→                except:
   646→                    pass
   647→
   648→                self._send_json({'success': True, 'message': f'Updated {len(resources)} resources'})
   649→
   650→            elif path == '/api/date':
   651→                if not save_state['gamestate_text']:
   652→                    self._send_json({'error': 'No save file loaded'}, 400)
   653→                    return
   654→
   655→                new_date = data.get('date', '')
   656→                if not re.match(r'^\d{4}\.\d{2}\.\d{2}$', new_date):
   657→                    self._send_json({'error': 'Invalid date format, use YYYY.MM.DD'}, 400)
   658→                    return
   659→
   660→                save_state['gamestate_text'] = modify_date_in_text(save_state['gamestate_text'], new_date)
   661→                save_state['meta_text'] = modify_date_in_text(save_state['meta_text'], new_date)
   662→
   663→                try:
   664→                    save_state['gamestate_parsed'] = parse_clausewitz(save_state['gamestate_text'])
   665→                    save_state['meta_parsed'] = parse_clausewitz(save_state['meta_text'])
   666→                except:
   667→                    pass
   668→
   669→                self._send_json({'success': True, 'date': new_date})
   670→
   671→            elif path == '/api/name':
   672→                if not save_state['gamestate_text']:
   673→                    self._send_json({'error': 'No save file loaded'}, 400)
   674→                    return
   675→
   676→                new_name = data.get('name', '')
   677→                old_name = save_state['meta_parsed'].get('name', '') if save_state['meta_parsed'] else ''
   678→
   679→                if old_name:
   680→                    save_state['gamestate_text'] = modify_name_in_text(save_state['gamestate_text'], old_name, new_name)
   681→                save_state['meta_text'] = modify_name_in_meta(save_state['meta_text'], new_name)
   682→
   683→                try:
   684→                    save_state['gamestate_parsed'] = parse_clausewitz(save_state['gamestate_text'])
   685→                    save_state['meta_parsed'] = parse_clausewitz(save_state['meta_text'])
   686→                except:
   687→                    pass
   688→
   689→                self._send_json({'success': True, 'name': new_name})
   690→
   691→            else:
   692→                self._send_json({'error': 'Not found'}, 404)
   693→
   694→        except Exception as e:
   695→            traceback.print_exc()
   696→            self._send_json({'error': str(e)}, 500)
