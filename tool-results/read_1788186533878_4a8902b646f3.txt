    75→def extract_save(filepath: str):
    76→    """Extract .sav (ZIP) and read meta + gamestate texts."""
    77→    with zipfile.ZipFile(filepath, 'r') as z:
    78→        names = z.namelist()
    79→        meta_text = z.read('meta').decode('utf-8') if 'meta' in names else ''
    80→        gamestate_text = z.read('gamestate').decode('utf-8') if 'gamestate' in names else ''
    81→    return meta_text, gamestate_text

    84→def find_player_country_id(gamestate: dict) -> int:
    85→    """Find the player's country index."""
    86→    player = gamestate.get('player')
    87→    if not player:
    88→        return 0
    89→    if isinstance(player, dict):
    90→        items = player.get(None, [])
    91→        if items and isinstance(items, list) and len(items) > 0:
    92→            first = items[0]
    93→            if isinstance(first, dict) and 'country' in first:
    94→                return int(first['country'])
    95→    return 0

   228→def _find_block_by_path(text_lines: list, path_steps: list) -> tuple:
   229→    """
   230→    Navigate through nested blocks to find the target block.
   231→    Returns (start_line_idx, end_line_idx, indent_level) or (-1,-1,-1) if not found.
   232→    """
   233→    depth = 0
   234→    step_idx = 0
   235→    current_step = path_steps[0]
   236→    block_start = -1
   237→
   238→    for i, line in enumerate(text_lines):
   239→        stripped = line.strip()
   240→
   241→        # Track depth via braces
   242→        for ch in stripped:
   243→            if ch == '{':
   244→                depth += 1
   245→            elif ch == '}':
   246→                depth -= 1
   247→
   248→        if step_idx < len(path_steps):
   249→ # Check if this line matches the current step
   250→            pattern = re.compile(rf'^\s*{re.escape(str(current_step))}\s*[=\{{]')
   251→            if pattern.match(line):
   252→                if step_idx == 0:
   253→                    block_start = i
   254→                step_idx += 1
   255→                if step_idx < len(path_steps):
   256→                    current_step = path_steps[step_idx]
   257→                else:
   258→                    # Found the target block
   259→                    # Return the block content range (after the opening brace)
   260→                    return (block_start, i, depth)
   261→
   262→    return (-1, -1, -1)

   265→def modify_resource_in_text(text: str, country_idx: int, resource_key: str, new_value: float) -> str:
   266→    """Modify a single resource value directly in the raw gamestate text."""
   267→    lines = text.split('\n')
   268→
   269→    # State machine to navigate: country={ 0={ ... modules={ standard_economy_module={ resources={ resource_key=VALUE } } } } }
   270→    state = 'seeking_country'
   271→    country_depth = 0
   272→    country_idx_found = False
   273→    module_depth = 0
   274→    econ_found = False
   275→    resource_block_depth = 0
   276→    in_resource_block = False
   277→    result_lines = []
   278→    modified = False
   279→
   280→    for line in lines:
   281→        if modified:
   282→            result_lines.append(line)
   283→            continue
   284→
   285→        stripped = line.strip()
   286→
   287→        # Count braces on this line
   288→        opens = stripped.count('{')
   289→        closes = stripped.count('}')
   290→
   291→        if state == 'seeking_country':
   292→            if stripped == 'country={':
   293→                state = 'in_country'
   294→                country_depth = 1
   295→            result_lines.append(line)
   296→            continue
   297→
   298→        if state == 'in_country':
   299→            country_depth += opens - closes
   300→
   301→            if country_depth <= 0:
   302→                state = 'seeking_country'
   303→                result_lines.append(line)
   304→                continue
   305→
   306→            if not country_idx_found:
   307→                # Look for the specific country index
   308→                idx_pat = re.compile(r'^\s*' + re.escape(str(country_idx)) + r'=\{\s*$')
   309→                if idx_pat.match(line):
   310→                    country_idx_found = True
   311→                    result_lines.append(line)
   312→                    continue
   313→
   314→            if country_idx_found and not econ_found:
   315→                # Look for standard_economy_module
   316→                if re.match(r'^\s*standard_economy_module\s*={\s*$', line):
   317→                    econ_found = True
   318→                    result_lines.append(line)
   319→                    continue
   320→
   321→            if econ_found and not in_resource_block:
   322→                if re.match(r'^\s*resources\s*={\s*$', line):
   323→                    in_resource_block = True
   324→                    resource_block_depth = 1
   325→                    result_lines.append(line)
   326→                    continue
   327→
   328→            if in_resource_block:
   329→                resource_block_depth += opens - closes
   330→
   331→                if resource_block_depth <= 0:
   332→                    in_resource_block = False
   333→                    result_lines.append(line)
   334→                    continue
   335→
   336→                # Check for the resource key
   337→                match = re.match(rf'^(\s*){re.escape(resource_key)}=(.+)$', line)
   338→                if match:
   339→                    indent = match.group(1)
   340→                    formatted = f'{new_value:.5f}'.rstrip('0')
   341→                    if formatted.endswith('.'):
   342→                        formatted += '0'
   343→                    result_lines.append(f'{indent}{resource_key}={formatted}')
   344→                    modified = True
   345→                    continue
   346→
   347→            result_lines.append(line)
   348→            continue
   349→
   350→        result_lines.append(line)
   351→
   352→    return '\n'.join(result_lines)

   361→def modify_name_in_text(text: str, old_name: str, new_name: str) -> str:
   362→    """Modify the empire name in gamestate text."""
   363→    # In gamestate, the name is in the first country's name block
   364→    # Format: name={\n\tkey="OLDNAME"\n}
   365→    pattern = re.compile(r'^(name=\{\n\s*key=")' + re.escape(old_name) + r'(")', re.MULTILINE)
   366→    result = pattern.sub(f'\\g<1>{new_name}\\g<2>', text, count=1)
   367→    return result

   457→def get_delayed_events_from_text(text, country_idx):
   458→    """Extract delayed events directly from raw text (fast, no full parse needed)."""
   459→    lines = text.split('\n')
   460→    state = 'seeking_country'
   461→    country_depth = 0
   462→    country_idx_found = False
   463→    econ_found = False
   464→    in_de = False
   465→    de_depth = 0
   466→    events = []
   467→    current_evt = {}
   468→    for line in lines:
   469→        stripped = line.strip()
   470→        opens = stripped.count('{')
   471→        closes = stripped.count('}')
   472→        if state == 'seeking_country':
   473→            if stripped == 'country={':
   474→                state = 'in_country'
   475→                country_depth = 1
   476→            continue
   477→        if state == 'in_country':
   478→            country_depth += opens - closes
   479→            if country_depth <= 0:
   480→                state = 'seeking_country'
   481→                continue
   482→            if not country_idx_found:
   483→                if re.match(r'^\s*' + re.escape(str(country_idx)) + r'=\{\s*$', line):
   484→                    country_idx_found = True
   485→                    continue
   486→            if country_idx_found and not econ_found:
   487→                if re.match(r'^\s*standard_economy_module\s*=\{\s*$', line):
   488→                    econ_found = True
   489→                    continue
   490→            if econ_found and not in_de:
   491→                if re.match(r'^\s*delayed_event\s*=\{\s*$', line):
   492→                    in_de = True
   493→                    de_depth = 0
   494→                    continue
   495→            if in_de:
   496→                de_depth += opens - closes
   497→                if de_depth <= 0:
   498→                    in_de = False
   499→                    continue
   500→                m = re.match(r'^\s*event="([^"]+)"', line)
   501→                if m:
   502→                    current_evt = {'event': m.group(1), 'days': 0}
   503→                    continue
   504→                m = re.match(r'^\s*days=(\d+)', line)
   505→                if m and current_evt:
   506→                    current_evt['days'] = int(m.group(1))
   507→                    events.append(current_evt)
   508→                    current_evt = {}
   509→                    continue
   510→    return events

   539→def modify_event_days_in_text(text: str, country_idx: int, event_index: int, new_days: int) -> str:
   540→    """Modify the days value of a specific delayed event in raw text."""
   541→    lines = text.split('\n')
   542→    state = 'seeking_country'
   543→    country_depth = 0
   544→    country_idx_found = False
   545→    econ_found = False
   546→    in_delayed_events = False
   547→    de_depth = 0
   548→    current_event_idx = -1
   549→    result_lines = []
   550→    modified = False
   551→
   552→    for line in lines:
   553→        if modified:
   554→            result_lines.append(line)
   555→            continue
   556→        stripped = line.strip()
   557→        opens = stripped.count('{')
   558→        closes = stripped.count('}')
   559→        if state == 'seeking_country':
   560→            if stripped == 'country={':
   561→                state = 'in_country'
   562→                country_depth = 1
   563→            result_lines.append(line)
   564→            continue
   565→        if state == 'in_country':
   566→            country_depth += opens - closes
   567→            if country_depth <= 0:
   568→                state = 'seeking_country'
   569→                result_lines.append(line)
   570→                continue
   571→            if not country_idx_found:
   572→                idx_pat = re.compile(r'\s*' + re.escape(str(country_idx)) + r'=\{\s*$', re.MULTILINE)
   573→                if idx_pat.match(line):
   574→                    country_idx_found = True
   575→                    result_lines.append(line)
   576→                    continue
   577→            if country_idx_found and not econ_found:
   578→                if re.match(r'^\s*standard_economy_module\s*=\{\s*$', line):
   579→                    econ_found = True
   580→                    result_lines.append(line)
   581→                    continue
   582→            if econ_found and not in_delayed_events:
   583→                if re.match(r'^\s*delayed_event\s*=\{\s*$', line):
   584→                    in_delayed_events = True
   585→                    de_depth = 0
   586→                    result_lines.append(line)
   587→                    continue
   588→            if in_delayed_events and not modified:
   589→                de_depth += opens - closes
   590→                if de_depth <= 0:
   591→                    in_delayed_events = False
   592→                    result_lines.append(line)
   593→                    continue
   594→                if de_depth == 1 and re.match(r'^\s*event=', line):
   595→                    current_event_idx += 1
   596→                if current_event_idx == event_index and re.match(r'^\s*days=', line):
   597→                    indent = line[:len(line) - len(line.lstrip())]
   598→                    result_lines.append(f'{indent}days={new_days}')
   599→                    modified = True
   600→                    continue
   601→            result_lines.append(line)
   602→            continue
   603→        result_lines.append(line)
   604→    return '\n'.join(result_lines)

   619→    def _send_file(self, filepath: str, filename: str):
   620→        import mimetypes
   621→        ctype = mimetypes.guess_type(filename)[0] or 'application/octet-stream'
   622→        size = os.path.getsize(filepath)
   623→        self.send_response(200)
   624→        self.send_header('Content-Type', ctype)
   625→        self.send_header('Content-Disposition', f'attachment; filename="{filename}"')
   626→        self.send_header('Access-Control-Allow-Origin', '*')
   627→        self.send_header('Content-Length', str(size))
   628→        self.end_headers()
   629→        with open(filepath, 'rb') as f:
   630→            while True:
   631→                chunk = f.read(65536)
   632→                if not chunk:
   633→                    break
   634→                self.wfile.write(chunk)
