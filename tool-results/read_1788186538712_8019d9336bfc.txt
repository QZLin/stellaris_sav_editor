    75→    75→def extract_save(filepath: str):
    76→    76→    """Extract .sav (ZIP) and read meta + gamestate texts."""
    77→    77→    with zipfile.ZipFile(filepath, 'r') as z:
    78→    78→        names = z.namelist()
    79→    79→        meta_text = z.read('meta').decode('utf-8') if 'meta' in names else ''
    80→    80→        gamestate_text = z.read('gamestate').decode('utf-8') if 'gamestate' in names else ''
    81→    81→    return meta_text, gamestate_text

   228→   228→def _find_block_by_path(text_lines: list, path_steps: list) -> tuple:
   229→   229→    """
   230→   230→    Navigate through nested blocks to find the target block.
   231→   231→    Returns (start_line_idx, end_line_idx, indent_level) or (-1,-1,-1) if not found.
   232→   232→    """
   233→   233→    depth = 0
   234→   234→    step_idx = 0
   235→   235→    current_step = path_steps[0]
   236→   236→    block_start = -1
   237→   237→
   238→   238→    for i, line in enumerate(text_lines):
   239→   239→        stripped = line.strip()
   240→   240→
   241→   241→        # Track depth via braces
   242→   242→        for ch in stripped:
   243→   243→            if ch == '{':
   244→   244→                depth += 1
   245→   245→            elif ch == '}':
   246→   246→                depth -= 1
   247→   247→
   248→   248→        if step_idx < len(path_steps):
   249→   249→ # Check if this line matches the current step
   250→   250→            pattern = re.compile(rf'^\s*{re.escape(str(current_step))}\s*[=\{{]')
   251→   251→            if pattern.match(line):
   252→   252→                if step_idx == 0:
   253→   253→                    block_start = i
   254→   254→                step_idx += 1
   255→   255→                if step_idx < len(path_steps):
   256→   256→                    current_step = path_steps[step_idx]
   257→   257→                else:
   258→   258→                    # Found the target block
   259→   259→                    # Return the block content range (after the opening brace)
   260→   260→                    return (block_start, i, depth)
   261→   261→
   262→   262→    return (-1, -1, -1)

   265→   265→def modify_resource_in_text(text: str, country_idx: int, resource_key: str, new_value: float) -> str:
   266→   266→    """Modify a single resource value directly in the raw gamestate text."""
   267→   267→    lines = text.split('\n')
   268→   268→
   269→   269→    # State machine to navigate: country={ 0={ ... modules={ standard_economy_module={ resources={ resource_key=VALUE } } } } }
   270→   270→    state = 'seeking_country'
   271→   271→    country_depth = 0
   272→   272→    country_idx_found = False
   273→   273→    module_depth = 0
   274→   274→    econ_found = False
   275→   275→    resource_block_depth = 0
   276→   276→    in_resource_block = False
   277→   277→    result_lines = []
   278→   278→    modified = False
   279→   279→
   280→   280→    for line in lines:
   281→   281→        if modified:
   282→   282→            result_lines.append(line)
   283→   283→            continue
   284→   284→
   285→   285→        stripped = line.strip()
   286→   286→
   287→   287→        # Count braces on this line
   288→   288→        opens = stripped.count('{')
   289→   289→        closes = stripped.count('}')
   290→   290→
   291→   291→        if state == 'seeking_country':
   292→   292→            if stripped == 'country={':
   293→   293→                state = 'in_country'
   294→   294→                country_depth = 1
   295→   295→            result_lines.append(line)
   296→   296→            continue
   297→   297→
   298→   298→        if state == 'in_country':
   299→   299→            country_depth += opens - closes
   300→   300→
   301→   301→            if country_depth <= 0:
   302→   302→                state = 'seeking_country'
   303→   303→                result_lines.append(line)
   304→   304→                continue
   305→   305→
   306→   306→            if not country_idx_found:
   307→   307→                # Look for the specific country index
   308→   308→                idx_pat = re.compile(r'^\s*' + re.escape(str(country_idx)) + r'=\{\s*$')
   309→   309→                if idx_pat.match(line):
   310→   310→                    country_idx_found = True
   311→   311→                    result_lines.append(line)
   312→   312→                    continue
   313→   313→
   314→   314→            if country_idx_found and not econ_found:
   315→   315→                # Look for standard_economy_module
   316→   316→                if re.match(r'^\s*standard_economy_module\s*={\s*$', line):
   317→   317→                    econ_found = True
   318→   318→                    result_lines.append(line)
   319→   319→                    continue
   320→   320→
   321→   321→            if econ_found and not in_resource_block:
   322→   322→                if re.match(r'^\s*resources\s*={\s*$', line):
   323→   323→                    in_resource_block = True
   324→   324→                    resource_block_depth = 1
   325→   325→                    result_lines.append(line)
   326→   326→                    continue
   327→   327→
   328→   328→            if in_resource_block:
   329→   329→                resource_block_depth += opens - closes
   330→   330→
   331→   331→                if resource_block_depth <= 0:
   332→   332→                    in_resource_block = False
   333→   333→                    result_lines.append(line)
   334→   334→                    continue
   335→   335→
   336→   336→                # Check for the resource key
   337→   337→                match = re.match(rf'^(\s*){re.escape(resource_key)}=(.+)$', line)
   338→   338→                if match:
   339→   339→                    indent = match.group(1)
   340→   340→                    formatted = f'{new_value:.5f}'.rstrip('0')
   341→   341→                    if formatted.endswith('.'):
   342→   342→                        formatted += '0'
   343→   343→                    result_lines.append(f'{indent}{resource_key}={formatted}')
   344→   344→                    modified = True
   345→   345→                    continue
   346→   346→
   347→   347→            result_lines.append(line)
   348→   348→            continue
   349→   349→
   350→   350→        result_lines.append(line)
   351→   351→
   352→   352→    return '\n'.join(result_lines)

   362→   362→    """Modify the empire name in gamestate text."""
   363→   363→    # In gamestate, the name is in the first country's name block
   364→   364→    # Format: name={\n\tkey="OLDNAME"\n}
   365→   365→    pattern = re.compile(r'^(name=\{\n\s*key=")' + re.escape(old_name) + r'(")', re.MULTILINE)
   366→   366→    result = pattern.sub(f'\\g<1>{new_name}\\g<2>', text, count=1)
   367→   367→    return result

   457→   457→def get_delayed_events_from_text(text, country_idx):
   458→   458→    """Extract delayed events directly from raw text (fast, no full parse needed)."""
   459→   459→    lines = text.split('\n')
   460→   460→    state = 'seeking_country'
   461→   461→    country_depth = 0
   462→   462→    country_idx_found = False
   463→   463→    econ_found = False
   464→   464→    in_de = False
   465→   465→    de_depth = 0
   466→   466→    events = []
   467→   467→    current_evt = {}
   468→   468→    for line in lines:
   469→   469→        stripped = line.strip()
   470→   470→        opens = stripped.count('{')
   471→   471→        closes = stripped.count('}')
   472→   472→        if state == 'seeking_country':
   473→   473→            if stripped == 'country={':
   474→   474→                state = 'in_country'
   475→   475→                country_depth = 1
   476→   476→            continue
   477→   477→        if state == 'in_country':
   478→   478→            country_depth += opens - closes
   479→   479→            if country_depth <= 0:
   480→   480→                state = 'seeking_country'
   481→   481→                continue
   482→   482→            if not country_idx_found:
   483→   483→                if re.match(r'^\s*' + re.escape(str(country_idx)) + r'=\{\s*$', line):
   484→   484→                    country_idx_found = True
   485→   485→                    continue
   486→   486→            if country_idx_found and not econ_found:
   487→   487→                if re.match(r'^\s*standard_economy_module\s*=\{\s*$', line):
   488→   488→                    econ_found = True
   489→   489→                    continue
   490→   490→            if econ_found and not in_de:
   491→   491→                if re.match(r'^\s*delayed_event\s*=\{\s*$', line):
   492→   492→                    in_de = True
   493→   493→                    de_depth = 0
   494→   494→                    continue
   495→   495→            if in_de:
   496→   496→                de_depth += opens - closes
   497→   497→                if de_depth <= 0:
   498→   498→                    in_de = False
   499→   499→                    continue
   500→   500→                m = re.match(r'^\s*event="([^"]+)"', line)
   501→   501→                if m:
   502→   502→                    current_evt = {'event': m.group(1), 'days': 0}
   503→   503→                    continue
   504→   504→                m = re.match(r'^\s*days=(\d+)', line)
   505→   505→                if m and current_evt:
   506→   506→                    current_evt['days'] = int(m.group(1))
   507→   507→                    events.append(current_evt)
   508→   508→                    current_evt = {}
   509→   509→                    continue
   510→   510→    return events

   539→   539→def modify_event_days_in_text(text: str, country_idx: int, event_index: int, new_days: int) -> str:
   540→   540→    """Modify the days value of a specific delayed event in raw text."""
   541→   541→    lines = text.split('\n')
   542→   542→    state = 'seeking_country'
   543→   543→    country_depth = 0
   544→   544→    country_idx_found = False
   545→   545→    econ_found = False
   546→   546→    in_delayed_events = False
   547→   547→    de_depth = 0
   548→   548→    current_event_idx = -1
   549→   549→    result_lines = []
   550→   550→    modified = False
   551→   551→
   552→   552→    for line in lines:
   553→   553→        if modified:
   554→   554→            result_lines.append(line)
   555→   555→            continue
   556→   556→        stripped = line.strip()
   557→   557→        opens = stripped.count('{')
   558→   558→        closes = stripped.count('}')
   559→   559→        if state == 'seeking_country':
   560→   560→            if stripped == 'country={':
   561→   561→                state = 'in_country'
   562→   562→                country_depth = 1
   563→   563→            result_lines.append(line)
   564→   564→            continue
   565→   565→        if state == 'in_country':
   566→   566→            country_depth += opens - closes
   567→   567→            if country_depth <= 0:
   568→   568→                state = 'seeking_country'
   569→   569→                result_lines.append(line)
   570→   570→                continue
   571→   571→            if not country_idx_found:
   572→   572→                idx_pat = re.compile(r'\s*' + re.escape(str(country_idx)) + r'=\{\s*$', re.MULTILINE)
   573→   573→                if idx_pat.match(line):
   574→   574→                    country_idx_found = True
   575→   575→                    result_lines.append(line)
   576→   576→                    continue
   577→   577→            if country_idx_found and not econ_found:
   578→   578→                if re.match(r'^\s*standard_economy_module\s*=\{\s*$', line):
   579→   579→                    econ_found = True
   580→   580→                    result_lines.append(line)
   581→   581→                    continue
   582→   582→            if econ_found and not in_delayed_events:
   583→   583→                if re.match(r'^\s*delayed_event\s*=\{\s*$', line):
   584→   584→                    in_delayed_events = True
   585→   585→                    de_depth = 0
   586→   586→                    result_lines.append(line)
   587→   587→                    continue
   588→   588→            if in_delayed_events and not modified:
   589→   589→                de_depth += opens - closes
   590→   590→                if de_depth <= 0:
   591→   591→                    in_delayed_events = False
   592→   592→                    result_lines.append(line)
   593→   593→                    continue
   594→   594→                if de_depth == 1 and re.match(r'^\s*event=', line):
   595→   595→                    current_event_idx += 1

   597→   597→                    indent = line[:len(line) - len(line.lstrip())]

   600→   600→                    continue
   601→   601→            result_lines.append(line)
   602→   602→            continue
   603→   603→        result_lines.append(line)
   604→   604→    return '\n'.join(result_lines)
