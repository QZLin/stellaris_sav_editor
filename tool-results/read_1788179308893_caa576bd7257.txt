    64→// ============ MAIN COMPONENT ============
    65→export default function StellarisSaveEditor() {
    66→  const { toast } = useToast();
    67→  const fileInputRef = useRef<HTMLInputElement>(null);
    68→
    69→  // State
    70→  const [loading, setLoading] = useState(false);
    71→  const [loaded, setLoaded] = useState(false);
    72→  const [meta, setMeta] = useState<SaveMeta | null>(null);
    73→  const [stats, setStats] = useState<GameStats | null>(null);
    74→  const [resources, setResources] = useState<Record<string, ResourceInfo>>({});
    75→  const [resourceCategories, setResourceCategories] = useState<ResourceCategory>({});
    76→  const [countries, setCountries] = useState<CountryInfo[]>([]);
    77→  const [species, setSpecies] = useState<SpeciesInfo[]>([]);
    78→  const [selectedCountry, setSelectedCountry] = useState('0');
    79→  const [editName, setEditName] = useState('');
    80→  const [editDate, setEditDate] = useState('');
    81→  const [editResources, setEditResources] = useState<Record<string, string>>({});
    82→  const [saving, setSaving] = useState(false);
    83→  const [exporting, setExporting] = useState(false);
    84→  const [activeTab, setActiveTab] = useState('overview');
    85→  const [events, setEvents] = useState<DelayedEvent[]>([]);
    86→  const [editEvents, setEditEvents] = useState<Record<number, string>>({});
    87→  const [flagData, setFlagData] = useState<FlagInfo | null>(null);
    88→  const [flagCategories, setFlagCategories] = useState<Record<string, FlagCategoryInfo>>({});
    89→  const [flagBgs, setFlagBgs] = useState<string[]>([]);
    90→  const [flagColors, setFlagColors] = useState<string[]>([]);
    91→
    92→  // Load save file
    93→  const handleFileUpload = useCallback(async (file: File) => {
    94→    if (!file.name.endsWith('.sav')) {
    95→      toast({ title: '错误', description: '请选择 .sav 格式的群星存档文件', variant: 'destructive' });
    96→      return;
    97→    }
    98→    setLoading(true);
    99→    try {
   100→      const result = await uploadSave(file);
   101→      setMeta(result.meta);
   102→      setEditName(result.meta.name);
   103→      setEditDate(result.meta.date);
   104→      setSelectedCountry(result.player_country_id);
   105→      setLoaded(true);
   106→      setActiveTab('overview');
   107→      toast({ title: '存档加载成功', description: `${result.meta.name} (${result.meta.date})` });
   108→
   109→      // Load additional data
   110→      const [statsData, countriesData, resourcesData, speciesData] = await Promise.all([
   111→        getStats(),
   112→        getCountries(),
   113→        getResources(result.player_country_id),
   114→        getSpecies(),
   115→      ]);
   116→      setStats(statsData);
   117→      setCountries(countriesData.countries);
   118→      setResources(resourcesData.resources);
   119→      setResourceCategories(resourcesData.categories);
   120→      setSpecies(speciesData.species);
   121→
   122→      // Init edit resources
   123→      const initEdit: Record<string, string> = {};
   124→      for (const [k, v] of Object.entries(resourcesData.resources)) {
   125→        initEdit[k] = v.value.toFixed(2);
   126→      }
   127→      setEditResources(initEdit);
   128→    } catch (err) {
   129→      toast({
   130→        title: '加载失败',
   131→        description: err instanceof Error ? err.message : '未知错误',
   132→        variant: 'destructive',
   133→      });
   134→    } finally {
   135→      setLoading(false);
   136→    }
   137→  }, [toast]);
   138→
   139→  // Handle drag & drop
   140→  const [dragOver, setDragOver] = useState(false);
   141→
   142→  const handleDrop = useCallback((e: React.DragEvent) => {
   143→    e.preventDefault();
   144→    setDragOver(false);
   145→    const file = e.dataTransfer.files[0];
   146→    if (file) handleFileUpload(file);
   147→  }, [handleFileUpload]);
   148→
   149→  // Save resources
   150→  const handleSaveResources = async () => {
   151→    setSaving(true);
   152→    try {
   153→      const parsed: Record<string, number> = {};
   154→      for (const [k, v] of Object.entries(editResources)) {
   155→        const num = parseFloat(v);
   156→        if (!isNaN(num)) parsed[k] = num;
   157→      }
   158→      await updateResources(selectedCountry, parsed);
   159→      // Refresh resources
   160→      const data = await getResources(selectedCountry);
   161→      setResources(data.resources);
   162→      const initEdit: Record<string, string> = {};
   163→      for (const [k, v] of Object.entries(data.resources)) {
   164→        initEdit[k] = v.value.toFixed(2);
   165→      }
   166→      setEditResources(initEdit);
   167→      toast({ title: '保存成功', description: '资源已更新' });
   168→    } catch (err) {
   169→      toast({ title: '保存失败', description: err instanceof Error ? err.message : '未知错误', variant: 'destructive' });
   170→    } finally {
   171→      setSaving(false);
   172→    }
   173→  };
   174→
   175→  // Save date
   176→  const handleSaveDate = async () => {
   177→    if (!/^\d{4}\.\d{2}\.\d{2}$/.test(editDate)) {
   178→      toast({ title: '格式错误', description: '日期格式应为 YYYY.MM.DD', variant: 'destructive' });
   179→      return;
   180→    }
   181→    setSaving(true);
   182→    try {
   183→      await updateDate(editDate);
   184→      const newStats = await getStats();
   185→      setStats(newStats);
   186→      if (meta) setMeta({ ...meta, date: editDate });
   187→      toast({ title: '日期已更新', description: editDate });
   188→    } catch (err) {
   189→      toast({ title: '更新失败', description: err instanceof Error ? err.message : '未知错误', variant: 'destructive' });
   190→    } finally {
   191→      setSaving(false);
   192→    }
   193→  };
   194→
   195→  // Save name
   196→  const handleSaveName = async () => {
   197→    if (!editName.trim()) {
   198→      toast({ title: '名称不能为空', variant: 'destructive' });
   199→      return;
   200→    }
   201→    setSaving(true);
   202→    try {
   203→      await updateName(editName.trim());
   204→      if (meta) setMeta({ ...meta, name: editName.trim() });
   205→      const newCountries = await getCountries();
   206→      setCountries(newCountries.countries);
   207→      toast({ title: '名称已更新', description: editName.trim() });
   208→    } catch (err) {
   209→      toast({ title: '更新失败', description: err instanceof Error ? err.message : '未知错误', variant: 'destructive' });
   210→    } finally {
   211→      setSaving(false);
   212→    }
   213→  };
   214→
   215→  // Export
   216→  const handleExport = async () => {
   217→    setExporting(true);
   218→    try {
   219→      const blob = await exportSave();
   220→      const url = URL.createObjectURL(blob);
   221→      const a = document.createElement('a');
   222→      a.href = url;
   223→      a.download = meta ? `${meta.name.replace(/[^a-zA-Z0-9\u4e00-\u9fa5]/g, '_')}_modified.sav` : 'modified_save.sav';
   224→      document.body.appendChild(a);
   225→      a.click();
   226→      document.body.removeChild(a);
   227→      URL.revokeObjectURL(url);
   228→      toast({ title: '导出成功', description: '修改后的存档已下载' });
   229→    } catch (err) {
   230→      toast({ title: '导出失败', description: err instanceof Error ? err.message : '未知错误', variant: 'destructive' });
   231→    } finally {
   232→      setExporting(false);
   233→    }
   234→  };
   235→
   236→  // Reset
   237→  const handleReset = async () => {
   238→    setLoaded(false);
   239→    setMeta(null);
   240→    setStats(null);
   241→    setResources({});
   242→    setCountries([]);
   243→    setSpecies([]);
   244→    setEditResources({});
   245→    try { await releaseSave(); } catch { /* ignore */ }
   246→    toast({ title: '存档已释放' });
   247→  };
   248→
   249→  // Switch country
   250→  const handleCountryChange = async (countryId: string) => {
   251→    setSelectedCountry(countryId);
   252→    try {
   253→      const data = await getResources(countryId);
   254→      setResources(data.resources);
   255→      setResourceCategories(data.categories);
   256→      const initEdit: Record<string, string> = {};
   257→      for (const [k, v] of Object.entries(data.resources)) {
   258→        initEdit[k] = v.value.toFixed(2);
   259→      }
   260→      setEditResources(initEdit);
   261→    } catch (err) {
   262→      toast({ title: '加载资源失败', description: err instanceof Error ? err.message : '', variant: 'destructive' });
   263→    }
   264→  };
   265→
   266→  // Quick set resource value
   267→  const setResourceValue = (key: string, value: string) => {
   268→    setEditResources(prev => ({ ...prev, [key]: value }));
   269→  };
   270→
   271→  // ============ RENDER: UPLOAD SCREEN ============
   272→  if (!loaded) {
   273→    return (
   274→      <div className="min-h-screen flex items-center justify-center p-4"
   275→        style={{ background: 'radial-gradient(ellipse at 20% 50%, rgba(30,10,60,0.8) 0%, rgba(5,5,15,1) 70%)' }}
   276→      >
   277→        <div className="absolute inset-0 overflow-hidden pointer-events-none">
   278→          {/* Stars background */}
   279→          {Array.from({ length: 80 }).map((_, i) => (
   280→            <div
   281→              key={i}
   282→              className="absolute rounded-full bg-white"
   283→              style={{
   284→                width: Math.random() * 2 + 1,
   285→                height: Math.random() * 2 + 1,
   286→                top: `${Math.random() * 100}%`,
   287→                left: `${Math.random() * 100}%`,
   288→                opacity: Math.random() * 0.7 + 0.3,
   289→                animation: `twinkle ${2 + Math.random() * 3}s ease-in-out ${Math.random() * 2}s infinite`,
   290→              }}
   291→            />
   292→          ))}
   293→        </div>
   294→
   295→        <Card className="relative z-10 w-full max-w-lg border-amber-500/20 bg-black/60 backdrop-blur-xl"
   296→          style={{ boxShadow: '0 0 80px rgba(245,158,11,0.1)' }}
   297→        >
   298→          <CardHeader className="text-center pb-2">
   299→            <div className="mx-auto mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-amber-500/10 border border-amber-500/30">
   300→              <Star className="h-8 w-8 text-amber-400" />
   301→            </div>
   302→            <CardTitle className="text-2xl font-bold text-amber-100">
   303→              群星存档修改器
   304→            </CardTitle>
   305→            <CardDescription className="text-amber-200/60 mt-1">
   306→              Stellaris Save Editor  ·  支持 Caelum / 最新版本存档
   307→            </CardDescription>
   308→          </CardHeader>
   309→          <CardContent>
   310→            <div
   311→              className={`border-2 border-dashed rounded-xl p-10 text-center transition-all cursor-pointer ${
   312→                dragOver
   313→                  ? 'border-amber-400 bg-amber-500/10'
   314→                  : 'border-gray-600 hover:border-amber-500/50 hover:bg-white/5'
   315→              }`}
   316→              onDragOver={(e) => { e.preventDefault(); setDragOver(true); }}
   317→              onDragLeave={() => setDragOver(false)}
   318→              onDrop={handleDrop}
   319→              onClick={() => fileInputRef.current?.click()}
   320→            >
   321→              <input
   322→                ref={fileInputRef}
   323→                type="file" accept=".sav" className="hidden"
   324→                onChange={(e) => {
   325→                  const file = e.target.files?.[0];
   326→                  if (file) handleFileUpload(file);
   327→                }}
   328→              />
   329→              {loading ? (
   330→                <div className="flex flex-col items-center gap-3">
   331→                  <Loader2 className="h-10 w-10 text-amber-400 animate-spin" />
   332→                  <p className="text-amber-200">正在解析存档，请稍候...</p>
   333→                  <p className="text-xs text-amber-200/40">大型存档可能需要10-20秒</p>
   334→                </div>
   335→              ) : (
   336→                <div className="flex flex-col items-center gap-3">
   337→                  <Upload className="h-10 w-10 text-amber-400/60" />
   338→                  <p className="text-amber-100 font-medium">拖拽 .sav 存档文件到此处</p>
   339→                  <p className="text-sm text-amber-200/50">或点击选择文件</p>
   340→                </div>
   341→              )}
   342→            </div>
   343→            <div className="mt-4 flex items-center gap-2 text-xs text-amber-200/30 justify-center">
   344→              <Shield className="h-3 w-3" />
   345→              <span>所有解析均在本地完成，不会上传任何数据</span>
   346→            </div>
   347→          </CardContent>
   348→        </Card>
   349→      </div>
   350→    );
   351→  }
   352→
   353→  // ============ RENDER: EDITOR SCREEN ============
   354→  return (
   355→    <div className="min-h-screen flex flex-col"
   356→      style={{ background: 'linear-gradient(135deg, rgba(5,5,20,1) 0%, rgba(15,10,30,1) 50%, rgba(5,10,25,1) 100%)' }}
   357→    >
   358→      {/* Header */}
   359→      <header className="sticky top-0 z-50 border-b border-amber-500/10 bg-black/70 backdrop-blur-xl">
   360→        <div className="max-w-7xl mx-auto px-4 h-14 flex items-center justify-between">
   361→          <div className="flex items-center gap-3">
   362→            <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-amber-500/15">
   363→              <Star className="h-4 w-4 text-amber-400" />
   364→            </div>
   365→            <div>
   366→              <h1 className="text-sm font-bold text-amber-100 leading-tight">{meta?.name || '群星存档修改器'}</h1>
   367→              <p className="text-xs text-amber-200/40">{meta?.version}  ·  {meta?.date}</p>
   368→            </div>
   369→            {meta?.ironman && (
   370→              <Badge variant="outline" className="border-red-500/40 text-red-400 text-xs">
   371→                铁人模式
   372→              </Badge>
   373→            )}
   374→          </div>
   375→          <div className="flex items-center gap-2">
   376→            <Button variant="outline" size="sm" onClick={handleReset}
   377→              className="border-gray-600 text-gray-300 hover:text-white hover:bg-white/5">
   378→              <RotateCcw className="h-3.5 w-3.5 mr-1" /> 释放存档
   379→            </Button>
   380→            <Button size="sm" onClick={handleExport} disabled={exporting}
   381→              className="bg-amber-500 hover:bg-amber-400 text-black font-medium">
   382→              {exporting ? <Loader2 className="h-3.5 w-3.5 mr-1 animate-spin" /> : <Download className="h-3.5 w-3.5 mr-1" />}
   383→              导出存档
   384→            </Button>
   385→          </div>
   386→        </div>
   387→      </header>
   388→
   389→      {/* Main Content */}
   390→      <main className="flex-1 max-w-7xl mx-auto w-full px-4 py-6">
   391→        <Tabs value={activeTab} onValueChange={setActiveTab}>
   392→          <TabsList className="bg-black/40 border border-amber-500/10 mb-6">
   393→            <TabsTrigger value="overview" className="data-[state=active]:bg-amber-500/20 data-[state=active]:text-amber-100">
   394→              <Globe className="h-3.5 w-3.5 mr-1.5" /> 总览
   395→            </TabsTrigger>
   396→            <TabsTrigger value="resources" className="data-[state=active]:bg-amber-500/20 data-[state=active]:text-amber-100">
   397→              <Zap className="h-3.5 w-3.5 mr-1.5" /> 资源
   398→            </TabsTrigger>
   399→            <TabsTrigger value="countries" className="data-[state=active]:bg-amber-500/20 data-[state=active]:text-amber-100">
   400→              <Flag className="h-3.5 w-3.5 mr-1.5" /> 国家
   401→            </TabsTrigger>
   402→            <TabsTrigger value="species" className="data-[state=active]:bg-amber-500/20 data-[state=active]:text-amber-100">
   403→              <Dna className="h-3.5 w-3.5 mr-1.5" /> 物种
   404→            </TabsTrigger>
   405→          </TabsList>
   406→
   407→          {/* ===== OVERVIEW TAB ===== */}
   408→          <TabsContent value="overview" className="space-y-6">
   409→            {/* Empire Info Card */}
   410→            <Card className="border-amber-500/10 bg-black/40 backdrop-blur">
   411→              <CardHeader className="pb-3">
   412→                <CardTitle className="text-amber-100 text-base flex items-center gap-2">
   413→                  <Star className="h-4 w-4" /> 帝国信息
   414→                </CardTitle>
   415→              </CardHeader>
   416→              <CardContent>
   417→                <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   418→                  {/* Name edit */}
   419→                  <div className="space-y-2">
   420→                    <Label className="text-amber-200/70 text-xs">帝国名称</Label>
   421→                    <div className="flex gap-2">
   422→                      <Input value={editName} onChange={e => setEditName(e.target.value)}
   423→                        className="bg-white/5 border-gray-600 text-amber-50" />
   424→                      <Button size="sm" onClick={handleSaveName} disabled={saving}
   425→                        className="bg-amber-500/20 hover:bg-amber-500/30 text-amber-200 border border-amber-500/30 shrink-0">
   426→                        <Save className="h-3.5 w-3.5" />
   427→                      </Button>
   428→                    </div>
   429→                  </div>
   430→                  {/* Date edit */}
   431→                  <div className="space-y-2">
   432→                    <Label className="text-amber-200/70 text-xs">游戏日期</Label>
   433→                    <div className="flex gap-2">
   434→                      <Input value={editDate} onChange={e => setEditDate(e.target.value)}
   435→                        placeholder="YYYY.MM.DD" className="bg-white/5 border-gray-600 text-amber-50" />
   436→                      <Button size="sm" onClick={handleSaveDate} disabled={saving}
   437→                        className="bg-amber-500/20 hover:bg-amber-500/30 text-amber-200 border border-amber-500/30 shrink-0">
   438→                        <Save className="h-3.5 w-3.5" />
   439→                      </Button>
   440→                    </div>
   441→                  </div>
   442→                </div>
   443→              </CardContent>
   444→            </Card>
   445→
   446→            {/* Stats Grid */}
   447→            {stats && (
   448→              <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   449→                {[
   450→                  { label: '游戏日期', value: stats.date, icon: <Calendar className="h-4 w-4 text-blue-400" /> },
   451→                  { label: '帝国规模', value: stats.empire_size, icon: <MapPin className="h-4 w-4 text-green-400" /> },
   452→                  { label: '科技数量', value: stats.tech_count, icon: <BookOpen className="h-4 w-4 text-purple-400" /> },
   453→                  { label: '舰队规模', value: stats.fleet_size, icon: <Rocket className="h-4 w-4 text-red-400" /> },
   454→                  { label: '军事力量', value: Math.round(stats.military_power).toLocaleString(), icon: <Swords className="h-4 w-4 text-orange-400" /> },
