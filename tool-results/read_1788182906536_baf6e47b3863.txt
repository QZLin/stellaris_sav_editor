    75→def extract_save(filepath: str):
    76→    """Extract .sav (ZIP) and read meta + gamestate texts."""
    77→    with zipfile.ZipFile(filepath, 'r') as z:
    78→        names = z.namelist()
    79→        meta_text = z.read('meta').decode('utf-8') if 'meta' in names else ''
    80→        gamestate_text = z.read('gamestate').decode('utf-8') if 'gamestate' in names else ''
    81→    return meta_text, gamestate_text

    86→    player = gamestate.get('player')
    87→    if not player:
    88→        return 0
    89→    if isinstance(player, dict):
    90→        items = player.get(None, [])
    91→        if items and isinstance(items, list) and len(items) > 0:
    92→            first = items[0]
    93→            if isinstance(first, dict) and 'country' in first:
    94→                return int(first['country'])

   228→def _find_block_by_path(text_lines: list, path_steps: list) -> tuple:
   229→    """
   230→    Navigate through nested blocks to find the target block.
   231→    Returns (start_line_idx, end_line_idx, indent_level) or (-1,-1,-1) if not found.
   232→    """
   233→    depth = 0
   234→    step_idx = 0
   235→    current_step = path_steps[0]
   236→    block_start = -1
   237→
   238→    for i, line in enumerate(text_lines):
   239→        stripped = line.strip()
   240→
   241→        # Track depth via braces
   242→        for ch in stripped:
   243→            if ch == '{':
   244→                depth += 1
   245→            elif ch == '}':
   246→                depth -= 1
   247→
   248→        if step_idx < len(path_steps):
   249→ # Check if this line matches the current step
   250→            pattern = re.compile(rf'^\s*{re.escape(str(current_step))}\s*[=\{{]')
   251→            if pattern.match(line):
   252→                if step_idx == 0:
   253→                    block_start = i
   254→                step_idx += 1
   255→                if step_idx < len(path_steps):
   256→                    current_step = path_steps[step_idx]
   257→                else:
   258→                    # Found the target block
   259→                    # Return the block content range (after the opening brace)
   260→                    return (block_start, i, depth)
   261→
   262→    return (-1, -1, -1)

   265→def modify_resource_in_text(text: str, country_idx: int, resource_key: str, new_value: float) -> str:
   266→    """Modify a single resource value directly in the raw gamestate text."""
   267→    lines = text.split('\n')
   268→
   269→    # State machine to navigate: country={ 0={ ... modules={ standard_economy_module={ resources={ resource_key=VALUE } } } } }
   270→    state = 'seeking_country'
   271→    country_depth = 0
   272→    country_idx_found = False
   273→    module_depth = 0
   274→    econ_found = False
   275→    resource_block_depth = 0
   276→    in_resource_block = False
   277→    result_lines = []
   278→    modified = False
   279→
   280→    for line in lines:
   281→        if modified:
   282→            result_lines.append(line)
   283→            continue
   284→
   285→        stripped = line.strip()
   286→
   287→        # Count braces on this line
   288→        opens = stripped.count('{')
   289→        closes = stripped.count('}')
   290→
   291→        if state == 'seeking_country':
   292→            if stripped == 'country={':
   293→                state = 'in_country'
   294→                country_depth = 1
   295→            result_lines.append(line)
   296→            continue
   297→
   298→        if state == 'in_country':
   299→            country_depth += opens - closes
   300→
   301→            if country_depth <= 0:
   302→                state = 'seeking_country'
   303→                result_lines.append(line)
   304→                continue
   305→
   306→            if not country_idx_found:
   307→                # Look for the specific country index
   308→                idx_pat = re.compile(r'^\s*' + re.escape(str(country_idx)) + r'=\{\s*$')
   309→                if idx_pat.match(line):
   310→                    country_idx_found = True
   311→                    result_lines.append(line)
   312→                    continue
   313→
   314→            if country_idx_found and not econ_found:
   315→                # Look for standard_economy_module
   316→                if re.match(r'^\s*standard_economy_module\s*={\s*$', line):
   317→                    econ_found = True
   318→                    result_lines.append(line)
   319→                    continue
   320→
   321→            if econ_found and not in_resource_block:
   322→                if re.match(r'^\s*resources\s*={\s*$', line):
   323→                    in_resource_block = True
   324→                    resource_block_depth = 1
   325→                    result_lines.append(line)
   326→                    continue
   327→
   328→            if in_resource_block:
   329→                resource_block_depth += opens - closes
   330→
   331→                if resource_block_depth <= 0:
   332→                    in_resource_block = False
   333→                    result_lines.append(line)
   334→                    continue
   335→
   336→                # Check for the resource key
   337→                match = re.match(rf'^(\s*){re.escape(resource_key)}=(.+)$', line)
   338→                if match:
   339→                    indent = match.group(1)
   340→                    formatted = f'{new_value:.5f}'.rstrip('0')
   341→                    if formatted.endswith('.'):
   342→                        formatted += '0'
   343→                    result_lines.append(f'{indent}{resource_key}={formatted}')
   344→                    modified = True
   345→                    continue
   346→
   347→            result_lines.append(line)
   348→            continue
   349→
   350→        result_lines.append(line)
   351→
   352→    return '\n'.join(result_lines)

   355→def modify_date_in_text(text: str, new_date: str) -> str:
   356→    """Modify the game date in text."""
   357→    pattern = re.compile(r'^(date=")\d{4}\.\d{2}\.\d{2}(")', re.MULTILINE)
   358→    return pattern.sub(f'\\g<1>{new_date}\\g<2>', text, count=1)

   361→def modify_name_in_text(text: str, old_name: str, new_name: str) -> str:
   362→    """Modify the empire name in gamestate text."""
   363→    # In gamestate, the name is in the first country's name block
   364→    # Format: name={\n\tkey="OLDNAME"\n}
   365→    pattern = re.compile(r'^(name=\{\n\s*key=")' + re.escape(old_name) + r'(")', re.MULTILINE)
   366→    result = pattern.sub(f'\\g<1>{new_name}\\g<2>', text, count=1)
   367→    return result

   513→                result_lines.append(line)
   514→                continue
   515→            if not country_idx_found:
   516→                idx_pat = re.compile(r'\s*' + re.escape(str(country_idx)) + r'=\{\s*$', re.MULTILINE)
   517→                if idx_pat.match(line):
   518→                    country_idx_found = True
   519→                    result_lines.append(line)
   520→                    continue
   521→            if country_idx_found and not econ_found:
   522→                if re.match(r'^\s*standard_economy_module\s*=\{\s*$', line):
   523→                    econ_found = True
   524→                    result_lines.append(line)
   525→                    continue
   526→            if econ_found and not in_delayed_events:
   527→                if re.match(r'^\s*delayed_event\s*=\{\s*$', line):
   528→                    in_delayed_events = True
   529→                    de_depth = 0
   530→                    result_lines.append(line)
   531→                    continue
   532→            if in_delayed_events and not modified:
   533→                de_depth += opens - closes
   534→                if de_depth <= 0:
   535→                    in_delayed_events = False
   536→                    result_lines.append(line)
   537→                    continue
   538→                if de_depth == 1 and re.match(r'^\s*event=', line):
   539→                    current_event_idx += 1
   540→                if current_event_idx == event_index and re.match(r'^\s*days=', line):
   541→                    indent = line[:len(line) - len(line.lstrip())]
   542→                    result_lines.append(f'{indent}days={new_days}')
   543→                    modified = True
   544→                    continue
   545→            result_lines.append(line)
   546→            continue
   547→        result_lines.append(line)
   548→    return '\n'.join(result_lines)

   572→        self.end_headers()

   574→            while True:
   575→                chunk = f.read(65536)
   576→                if not chunk:
   577→                    break
   578→                self.wfile.write(chunk)
