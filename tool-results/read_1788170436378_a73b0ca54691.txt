    56→def extract_save(filepath: str):
    57→    """Extract .sav (ZIP) and read meta + gamestate texts."""
    58→    with zipfile.ZipFile(filepath, 'r') as z:
    59→        names = z.namelist()
    60→        meta_text = z.read('meta').decode('utf-8') if 'meta' in names else ''
    61→        gamestate_text = z.read('gamestate').decode('utf-8') if 'gamestate' in names else ''
    62→    return meta_text, gamestate_text

    65→def get_nested(data: dict, path: list):
    66→    """Get a nested value from parsed data using a path like ['country', 0, 'resources']."""
    67→    current = data
    68→    for key in path:
    69→        if isinstance(current, dict):
    70→            if key in current:
    71→                current = current[key]
    72→            else:
    73→                return None
    74→        elif isinstance(current, list):
    75→            try:
    76→                idx = int(key)
    77→                current = current[idx]
    78→            except (ValueError, IndexError):
    79→                return None
    80→        else:
    81→            return None
    82→    return current

    88→        return
    89→    current = data
    90→    for key in path[:-1]:
    91→        if isinstance(current, dict) and key in current:
    92→            current = current[key]
    93→        elif isinstance(current, list):
    94→            try:
    95→                current = current[int(key)]
    96→            except (ValueError, IndexError):
    97→                return
    98→        else:
    99→            return
   100→    if isinstance(current, dict):
   101→        current[path[-1]] = value

   104→def find_country_player_index(gamestate: dict) -> int:
   105→    """Find the player's country index from the player block."""
   106→    player = gamestate.get('player')
   107→    if not player:
   108→        return 0
   109→    # player is a list of dicts with 'country' key
   110→    if isinstance(player, list):
   111→        for p in player:
   112→            if isinstance(p, dict) and 'country' in p:
   113→                return p['country']
   114→        if player and isinstance(player[0], dict) and 'country' in player[0]:
   115→            return player[0]['country']
   116→    return 0

   119→def get_country_resources(gamestate: dict, country_idx: int) -> dict:
   120→    """Extract resources for a specific country."""
   121→    countries = gamestate.get('country')
   122→    if not countries or not isinstance(countries, dict):
   123→        return {}
   124→    c_key = str(country_idx)
   125→    if c_key not in countries:
   126→        # Try int key
   127→        if country_idx in countries:
   128→            c_key = country_idx
   129→        else:
   130→            return {}
   131→    country = countries[c_key]
   132→    if not isinstance(country, dict):
   133→        return {}
   134→    resources = country.get('resources', {})
   135→    return resources if isinstance(resources, dict) else {}

   187→def modify_text_value(text: str, path: list, old_value: str, new_value: str) -> str:
   188→    """
   189→    Directly modify a value in the raw text by finding the key path and replacing.
   190→    This is more reliable for large files than full parse-modify-serialize.
   191→    Uses a state-machine approach to navigate braces.
   192→    """
   193→    # For resources, use a targeted approach
   194→    if len(path) >= 2 and path[0] == 'country' and 'resources' in path:
   195→        return _modify_in_block(text, path, old_value, new_value)
   196→    # For date
   197→    if path == ['date']:
   198→        pattern = re.compile(r'^date="[^"]*"', re.MULTILINE)
   199→        return pattern.sub(f'date="{new_value}"', text, count=1)
   200→    # For name
   201→    if path == ['name']:
   202→        pattern = re.compile(r'^name="[^"]*"', re.MULTILINE)
   203→        return pattern.sub(f'name="{new_value}"', text, count=1)
   204→    return text

   207→def _modify_in_block(text: str, path: list, old_value: str, new_value: str) -> str:
   208→    """Navigate into nested blocks and modify a specific key=value."""
   209→    lines = text.split('\n')
   210→    target_key = path[-1]
   211→    target_depth = len(path) - 1  # depth relative to top level
   212→
   213→    # We need to track brace depth
   214→    current_depth = 0
   215→    path_idx = 0
   216→    in_target_block = False
   217→    modifications = 0
   218→
   219→    # Build the path keys to match
   220→    path_keys = []
   221→    for p in path[:-1]:
   222→        if isinstance(p, str):
   223→            path_keys.append(p)
   224→
   225→    result_lines = []
   226→    i = 0
   227→    while i < len(lines):
   228→        line = lines[i]
   229→        stripped = line.strip()
   230→
   231→        # Count braces on this line
   232→        open_braces = stripped.count('{')
   233→        close_braces = stripped.count('}')
   234→
   235→        if in_target_block and modifications < 1:
   236→            # Check if this line has the target key
   237→            match = re.match(rf'^\s*{re.escape(target_key)}=(.+)$', line)
   238→            if match:
   239→                old_val = match.group(1).strip()
   240→                indent = line[:len(line) - len(line.lstrip())]
   241→                if isinstance(new_value, float):
   242→                    formatted = f'{new_value:.5f}'.rstrip('0')
   243→                    if formatted.endswith('.'):
   244→                        formatted += '0'
   245→                else:
   246→                    formatted = str(new_value)
   247→                result_lines.append(f'{indent}{target_key}={formatted}')
   248→                modifications += 1
   249→                i += 1
   250→                continue
   251→
   252→        # Check if we're entering the right block
   253→        if path_idx < len(path_keys):
   254→            key_match = re.match(rf'^\s*{re.escape(path_keys[path_idx])}\s*[=\{{]', line)
   255→            if key_match:
   256→                if open_braces > close_braces:
   257→                    current_depth += (open_braces - close_braces)
   258→                    path_idx += 1
   259→                    if path_idx >= len(path_keys):
   260→                        in_target_block = True
   261→                    result_lines.append(line)
   262→                    i += 1
   263→                    continue
   264→
   265→        if open_braces > 0:
   266→            current_depth += open_braces
   267→        if close_braces > 0:
   268→            current_depth -= close_braces
   269→            if current_depth <= 0 and path_idx < len(path_keys):
   270→                path_idx = 0
   271→                in_target_block = False
   272→
   273→        result_lines.append(line)
   274→        i += 1
   275→
   276→    return '\n'.join(result_lines)

   287→def _modify_resource_value(text: str, country_idx: int, resource_key: str, new_value: float) -> str:
   288→    """Modify a single resource value in the gamestate text."""
   289→    lines = text.split('\n')
   290→
   291→    # Find country block
   292→    country_pattern = f'country={{'
   293→    in_country = False
   294→    country_depth = 0
   295→    found_country_idx = None
   296→    in_resources = False
   297→    resource_depth = 0
   298→
   299→    result_lines = []
   300→    i = 0
   301→    while i < len(lines):
   302→        line = lines[i]
   303→        stripped = line.strip()
   304→
   305→        if not in_country:
   306→            if stripped == 'country={':
   307→                in_country = True
   308→                country_depth = 1
   309→                result_lines.append(line)
   310→                i += 1
   311→                continue
   312→            result_lines.append(line)
   313→            i += 1
   314→                continue
   315→
   316→        # We're inside country block
   317→        # Track depth
   318→        for ch in stripped:
   319→            if ch == '{':
   320→                country_depth += 1
   321→            elif ch == '}':
   322→                country_depth -= 1
   323→
   324→        if country_depth <= 0:
   325→            in_country = False
   326→            in_resources = False
   327→            result_lines.append(line)
   328→            i += 1
   329→            continue
   330→
   331→        # Check for country index
   332→        if found_country_idx is None and stripped.startswith(str(country_idx) + '='):
   333→            found_country_idx = True
   334→
   335→        # Check for resources block
   336→        if found_country_idx is not None:
   337→            if stripped == 'resources={':
   338→                in_resources = True
   339→                resource_depth = 1
   340→                result_lines.append(line)
   341→                i += 1
   342→                continue
   343→
   344→        if in_resources:
   345→            for ch in stripped:
   346→                if ch == '{':
   347→                    resource_depth += 1
   348→                elif ch == '}':
   349→                    resource_depth -= 1
   350→
   351→            if resource_depth <= 0:
   352→                in_resources = False
   353→                result_lines.append(line)
   354→                i += 1
   355→                continue
   356→
   357→            # Check for resource key
   358→            match = re.match(rf'^\s*{re.escape(resource_key)}=(.+)$', line)
   359→            if match:
   360→                indent = line[:len(line) - len(line.lstrip())]
   361→                formatted = f'{new_value:.5f}'.rstrip('0')
   362→                if formatted.endswith('.'):
   363→                    formatted += '0'
   364→                result_lines.append(f'{indent}{resource_key}={formatted}')
   365→                i += 1
   366→                continue
   367→
   368→        result_lines.append(line)
   369→        i += 1
   370→
   371→    return '\n'.join(result_lines)

   380→def modify_name(text: str, new_name: str) -> str:
   381→    """Modify the empire name."""
   382→    # The name appears in gamestate as name={key="..."}
   383→    # We need to find the first occurrence after country={0={
   384→    lines = text.split('\n')
   385→    result_lines = []
   386→    in_player_country = False
   387→    depth = 0
   388→    name_found = False
   389→
   390→    for line in lines:
   391→        if name_found:
   392→            result_lines.append(line)
   393→            continue
   394→
   395→        stripped = line.strip()
   396→
   397→        if not in_player_country:
   398→            if 'country={' in stripped and stripped.startswith('country'):
   399→                # Check if next non-empty line starts with 0=
   400→                pass
   401→            if re.match(r'^\d+=\{$', stripped):
   402→                in_player_country = True
   403→                depth = 1
   404→                result_lines.append(line)
   405→                continue
   406→
   407→        if in_player_country:
   408→            for ch in stripped:
   409→                if ch == '{':
   410→                    depth += 1
   411→                elif ch == '}':
   412→                    depth -= 1
   413→
   414→            if depth <= 0:
   415→                in_player_country = False
   416→                result_lines.append(line)
   417→                continue
   418→
   419→            if 'name={' in stripped:
   420→                indent = line[:len(line) - len(line.lstrip())]
   421→                result_lines.append(f'{indent}name={{')
   422→                result_lines.append(f'{indent}\tkey="{new_name}"')
   423→                result_lines.append(f'{indent}}}')
   424→                name_found = True
   425→                continue
   426→
   427→        result_lines.append(line)
   428→
   429→    return '\n'.join(result_lines)

   432→class SaveHandler(BaseHTTPRequestHandler):
   433→    """HTTP request handler for save file operations."""
   434→
   435→    def _send_json(self, data: dict, status: int = 200):
   436→        body = json.dumps(data, ensure_ascii=False).encode('utf-8')
   437→        self.send_response(status)
   438→        self.send_header('Content-Type', 'application/json; charset=utf-8')
   439→        self.send_header('Content-Length', str(len(body)))
   440→        self.end_headers()
   441→        self.wfile.write(body)
   442→
   443→    def _send_file(self, filepath: str, filename: str):
   444→        import mimetypes
   445→        ctype = mimetypes.guess_type(filename)[0] or 'application/octet-stream'
   446→        size = os.path.getsize(filepath)
   447→        self.send_response(200)
   448→        self.send_header('Content-Type', ctype)
   449→        self.send_header('Content-Disposition', f'attachment; filename="{filename}"')
   450→        self.send_header('Content-Length', str(size))
   451→        self.end_headers()
   452→        with open(filepath, 'rb') as f:
   453→            while True:
   454→                chunk = f.read(65536)
   455→                if not chunk:
   456→                    break
   457→                self.wfile.write(chunk)
   458→
   459→    def _read_body(self) -> bytes:
   460→        length = int(self.headers.get('Content-Length', 0))
   461→        return self.rfile.read(length) if length > 0 else b''
   462→
   463→    def do_OPTIONS(self):
   464→        self.send_response(200)
   465→        self.send_header('Access-Control-Allow-Origin', '*')
   466→        self.send_header('Access-Control-Allow-Methods', 'GET, POST, PUT, DELETE, OPTIONS')
   467→        self.send_header('Access-Control-Allow-Headers', 'Content-Type, X-Requested-With')
   468→        self.end_headers()
   469→
   470→    def do_GET(self):
   471→        parsed = urlparse(self.path)
   472→        path = parsed.path
   473→        params = parse_qs(parsed.query)
   474→
   475→        self.send_header('Access-Control-Allow-Origin', '*')
   476→
   477→        try:
   478→            if path == '/api/status':
   479→                loaded = save_state['sav_path'] is not None
   480→                self._send_json({
   481→                    'loaded': loaded,
   482→                    'filename': os.path.basename(save_state['sav_path']) if loaded else None,
   483→                })
   484→
   485→            elif path == '/api/meta':
   486→                if not save_state['meta_parsed']:
   487→                    self._send_json({'error': 'No save file loaded'}, 400)
   488→                    return
   489→                meta = save_state['meta_parsed']
   490→                self._send_json({
   491→                    'version': meta.get('version', ''),
   492→                    'name': meta.get('name', ''),
   493→                    'date': meta.get('date', ''),
   494→                    'ironman': meta.get('ironman', False),
   495→                    'dlcs': meta.get('required_dlcs', []),
   496→                    'meta_fleets': meta.get('meta_fleets', 0),
   497→                    'meta_planets': meta.get('meta_planets', 0),
   498→                })
   499→
   500→            elif path == '/api/countries':
   501→                if not save_state['gamestate_parsed']:
   502→                    self._send_json({'error': 'No save file loaded'}, 400)
   503→                    return
   504→                countries = get_countries_list(save_state['gamestate_parsed'])
   505→                player_idx = find_country_player_index(save_state['gamestate_parsed'])
   506→                self._send_json({
   507→                    'countries': countries,
   508→                    'player_country_id': player_idx,
   509→                })
   510→
   511→            elif path == '/api/resources':
   512→                if not save_state['gamestate_parsed']:
   513→                    self._send_json({'error': 'No save file loaded'}, 400)
   514→                    return
   515→                country_id = int(params.get('country_id', [0])[0])
   516→                resources = get_country_resources(save_state['gamestate_parsed'], country_id)
   517→                labeled = {}
   518→                for key in RESOURCE_KEYS:
   519→                    val = resources.get(key, 0)
   520→                    labeled[key] = {
   521→                        'value': float(val) if val else 0.0,
   522→                        'label': RESOURCE_LABELS.get(key, key),
   523→                    }
   524→                self._send_json({'resources': labeled, 'country_id': country_id})
   525→
   526→            elif path == '/api/species':
   527→                if not save_state['gamestate_parsed']:
   528→                    self._send_json({'error': 'No save file loaded'}, 400)
   529→                    return
   530→                species = get_species_list(save_state['gamestate_parsed'])
   531→                # Only return first 50 to avoid huge response
   532→                self._send_json({'species': species[:50], 'total': len(species)})
   533→
   534→            elif path == '/api/date':
   535→                if not save_state['gamestate_parsed']:
   536→                    self._send_json({'error': 'No save file loaded'}, 400)
   537→                    return
   538→                date = save_state['gamestate_parsed'].get('date', '')
   539→                self._send_json({'date': date})
   540→
   541→            elif path == '/api/gamestate-stats':
   542→                if not save_state['gamestate_parsed']:
   543→                    self._send_json({'error': 'No save file loaded'}, 400)
   544→                    return
   545→                gs = save_state['gamestate_parsed']
   546→                stats = {
   547→                    'date': gs.get('date', ''),
   548→                    'tick': gs.get('tick', 0),
   549→                    'num_species': len(gs.get('species_db', {})) if isinstance(gs.get('species_db'), dict) else 0,
   550→                    'num_countries': len(gs.get('country', {})) if isinstance(gs.get('country'), dict) else 0,
   551→                }
   552→                self._send_json(stats)
   553→
   554→            else:
   555→                self._send_json({'error': 'Not found'}, 404)
   556→
   557→        except Exception as e:
   558→            traceback.print_exc()
   559→            self._send_json({'error': str(e)}, 500)
   560→
   561→    def do_POST(self):
   562→        parsed = urlparse(self.path)
   563→        path = parsed.path
   564→
   565→        self.send_header('Access-Control-Allow-Origin', '*')
   566→
   567→        try:
   568→            if path == '/api/upload':
   569→                body = self._read_body()
   570→                if not body:
   571→                    self._send_json({'error': 'No file data'}, 400)
   572→                    return
   573→
   574→                # Save to temp file
   575→                tmp = tempfile.NamedTemporaryFile(suffix='.sav', delete=False)
   576→                tmp.write(body)
   577→                tmp.close()
   578→
   579→                # Extract
   580→                try:
   581→                    meta_text, gamestate_text = extract_save(tmp.name)
   582→                except Exception as e:
   583→                    os.unlink(tmp.name)
   584→                    self._send_json({'error': f'Failed to parse .sav file: {str(e)}'}, 400)
   585→                    return
   586→
   587→                # Parse meta
   588→                meta_parsed = parse_clausewitz(meta_text) if meta_text else {}
   589→
   590→                # Parse gamestate (this may take a moment for large files)
   591→                gamestate_parsed = None
   592→                try:
   593→                    gamestate_parsed = parse_clausewitz(gamestate_text) if gamestate_text else {}
   594→                except Exception as e:
   595→                    print(f'Warning: Full gamestate parse failed: {e}')
   596→                    # Store raw text for direct text manipulation
   597→                    gamestate_parsed = {}
   598→
   599→                save_state['sav_path'] = tmp.name
   600→                save_state['meta_text'] = meta_text
   601→                save_state['gamestate_text'] = gamestate_text
   602→                save_state['meta_parsed'] = meta_parsed
   603→                save_state['gamestate_parsed'] = gamestate_parsed
   604→                save_state['modifications'] = {}
   605→
   606→                player_idx = find_country_player_index(gamestate_parsed) if gamestate_parsed else 0
   607→                meta_name = meta_parsed.get('name', '')
   608→                meta_date = meta_parsed.get('date', '')
   609→
   610→                self._send_json({
   611→                    'success': True,
   612→                    'filename': os.path.basename(tmp.name),
   613→                    'meta': {
   614→                        'version': meta_parsed.get('version', ''),
   615→                        'name': meta_name,
   616→                        'date': meta_date,
   617→                        'ironman': meta_parsed.get('ironman', False),
   618→                        'dlcs': meta_parsed.get('required_dlcs', []),
   619→                        'meta_fleets': meta_parsed.get('meta_fleets', 0),
   620→                        'meta_planets': meta_parsed.get('meta_planets', 0),
   621→                    },
   622→                    'player_country_id': player_idx,
   623→                    'gamestate_size': len(gamestate_text),
   624→                })
   625→
   626→            else:
   627→                self._send_json({'error': 'Not found'}, 404)
   628→
   629→        except Exception as e:
   630→            traceback.print_exc()
   631→            self._send_json({'error': str(e)}, 500)

   633→    def do_PUT(self):
   634→        parsed = urlparse(self.path)
   635→        path = parsed.path
   636→
   637→        self.send_header('Access-Control-Allow-Origin', '*')
   638→
   639→        try:
   640→            body = self._read_body()
   641→            data = json.loads(body) if body else {}
   642→
   643→            if path == '/api/resources':
   644→                if not save_state['gamestate_text']:
   645→                    self._send_json({'error': 'No save file loaded'}, 400)
   646→                    return
   647→
   648→                country_id = data.get('country_id', 0)
   649→                resources = data.get('resources', {})
   650→
   651→                # Apply changes to raw text
   652→                gs_text = save_state['gamestate_text']
   653→                for key, value in resources.items():
   654→                    try:
   655→                        gs_text = _modify_resource_value(gs_text, country_id, key, float(value))
   656→                    except (ValueError, TypeError):
   657→                        pass
   658→
   659→                save_state['gamestate_text'] = gs_text
   660→                # Re-parse the modified text
   661→                try:
   662→                    save_state['gamestate_parsed'] = parse_clausewitz(gs_text)
   663→                except:
   664→                    pass
   665→
   666→                self._send_json({'success': True, 'message': f'Updated {len(resources)} resources'})
   667→
   668→            elif path == '/api/date':
   669→                if not save_state['gamestate_text']:
   670→                    self._send_json({'error': 'No save file loaded'}, 400)
   671→                    return
   672→
   673→                new_date = data.get('date', '')
   674→                if not re.match(r'^\d{4}\.\d{2}\.\d{2}$', new_date):
   675→                    self._send_json({'error': 'Invalid date format, use YYYY.MM.DD'}, 400)
   676→                    return
   677→
   678→                gs_text = modify_date(save_state['gamestate_text'], new_date)
   679→                meta_text = modify_date(save_state['meta_text'], new_date)
   680→                save_state['gamestate_text'] = gs_text
   681→                save_state['meta_text'] = meta_text
   682→
   683→                try:
   684→                    save_state['gamestate_parsed'] = parse_clausewitz(gs_text)
   685→                    save_state['meta_parsed'] = parse_clausewitz(meta_text)
   686→                except:
   687→                    pass
   688→
   689→                self._send_json({'success': True, 'date': new_date})
   690→
   691→            elif path == '/api/name':
   692→                if not save_state['gamestate_text']:
   693→                    self._send_json({'error': 'No save file loaded'}, 400)
   694→                    return
   695→
   696→                new_name = data.get('name', '')
   697→                gs_text = modify_name(save_state['gamestate_text'], new_name)
   698→                meta_text = modify_name(save_state['meta_text'], new_name)  # meta uses simple format
   699→
   700→                # meta name is simpler: name="..."
   701→                meta_text = re.sub(r'^name="[^"]*"', f'name="{new_name}"', meta_text, count=1, flags=re.MULTILINE)
   702→
   703→                save_state['gamestate_text'] = gs_text
   704→                save_state['meta_text'] = meta_text
   705→
   706→                try:
   707→                    save_state['gamestate_parsed'] = parse_clausewitz(gs_text)
   708→                    save_state['meta_parsed'] = parse_clausewitz(meta_text)
   709→                except:
   710→                    pass
   711→
   712→                self._send_json({'success': True, 'name': new_name})
   713→
   714→            else:
   715→                self._send_json({'error': 'Not found'}, 404)
   716→
   717→        except Exception as e:
   718→            traceback.print_exc()
   719→            self._send_json({'error': str(e)}, 500)

   721→    def do_DELETE(self):
   722→        parsed = urlparse(self.path)
   723→        path = parsed.path
   724→
   725→        self.send_header('Access-Control-Allow-Origin', '*')
   726→

   734→                save_state['meta_parsed'] = None
   735→                save_state['gamestate_parsed'] = None
   736→                save_state['modifications'] = {}
   737→                self._send_json({'success': True})
   738→            else:
   739→                self._send_json({'error': 'Not found'}, 404)
   740→        except Exception as e:
   741→            self._send_json({'error': str(e)}, 500)
